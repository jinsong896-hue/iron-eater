### 2. 初始房代码

把下面脚本挂在 `StartRoom_01` 根节点：

```gdscript
extends Node2D

class_name StartRoom01


# =========================================================
# 《噬铁者》第一层初始房
# StartRoom_01
#
# 房间规格：
# 30 x 18 Tile
# Tile：64 x 64 px
# 墙体厚度：2 Tile
#
# 可活动区：
# 26 x 14 Tile
#
# 功能：
# - 初始安全区
# - 中央出生
# - 中央暖色光照
# - 四周草地
# - 南侧入口
# - 北侧出口
# =========================================================


@export_category("Room Size")

@export var room_width: int = 30
@export var room_height: int = 18
@export var wall_thickness: int = 2


@export_category("TileSet")

# TileSet 中的 Source ID
@export var tile_source_id: int = 0


# ---------- 地板 ----------

@export var floor_tile: Vector2i = Vector2i(0, 0)


# ---------- 墙 ----------

@export var wall_tile: Vector2i = Vector2i(1, 0)


# ---------- 草 ----------

@export var grass_tiles: Array[Vector2i] = [
	Vector2i(0, 1),
	Vector2i(1, 1),
	Vector2i(2, 1),
	Vector2i(3, 1)
]


@export_category("Grass")

# 草地向房间内部延伸多少格
@export_range(1, 6)
var grass_depth: int = 4


# 草地密度
@export_range(0.0, 1.0)
var grass_density: float = 0.65


# 中心区域保持无草
@export var center_clear_radius: float = 5.0


@export_category("Lighting")

@export var center_light_energy: float = 1.4

@export var center_light_scale: float = 5.0


# =========================================================
# Nodes
# =========================================================

@onready var ground: TileMapLayer = $Ground
@onready var grass: TileMapLayer = $Grass
@onready var walls: TileMapLayer = $Walls

@onready var center_light: PointLight2D = $CenterLight

@onready var player_spawn: Marker2D = $PlayerSpawn
@onready var entrance_south: Marker2D = $EntranceSouth
@onready var exit_north: Marker2D = $ExitNorth


var rng := RandomNumberGenerator.new()


# =========================================================
# Ready
# =========================================================

func _ready() -> void:

	rng.randomize()

	generate_room()

	setup_spawn_points()

	setup_light()


# =========================================================
# Generate Room
# =========================================================

func generate_room() -> void:

	ground.clear()
	grass.clear()
	walls.clear()

	generate_floor()
	generate_walls()
	generate_grass()


# =========================================================
# Floor
# =========================================================

func generate_floor() -> void:

	for x in range(room_width):

		for y in range(room_height):

			var cell := Vector2i(x, y)

			ground.set_cell(
				cell,
				tile_source_id,
				floor_tile
			)


# =========================================================
# Walls
# =========================================================

func generate_walls() -> void:

	for x in range(room_width):

		for y in range(room_height):

			var is_left_wall: bool = x < wall_thickness
			var is_right_wall: bool = x >= room_width - wall_thickness

			var is_top_wall: bool = y < wall_thickness
			var is_bottom_wall: bool = y >= room_height - wall_thickness


			if (
				is_left_wall
				or is_right_wall
				or is_top_wall
				or is_bottom_wall
			):

				var cell := Vector2i(x, y)

				walls.set_cell(
					cell,
					tile_source_id,
					wall_tile
				)


	# -----------------------------------------------------
	# 南侧入口
	# 开 3 Tile 宽
	# -----------------------------------------------------

	var center_x := room_width / 2

	for x in range(center_x - 1, center_x + 2):

		for y in range(
			room_height - wall_thickness,
			room_height
		):

			walls.erase_cell(
				Vector2i(x, y)
			)


	# -----------------------------------------------------
	# 北侧出口
	# 开 3 Tile 宽
	# -----------------------------------------------------

	for x in range(center_x - 1, center_x + 2):

		for y in range(0, wall_thickness):

			walls.erase_cell(
				Vector2i(x, y)
			)


# =========================================================
# Grass
# =========================================================

func generate_grass() -> void:

	if grass_tiles.is_empty():
		return


	var center := Vector2(
		room_width / 2.0,
		room_height / 2.0
	)


	for x in range(
		wall_thickness,
		room_width - wall_thickness
	):

		for y in range(
			wall_thickness,
			room_height - wall_thickness
		):

			var cell := Vector2i(x, y)


			# ---------------------------------------------
			# 与墙的距离
			# ---------------------------------------------

			var distance_left := x - wall_thickness

			var distance_right := (
				room_width
				- wall_thickness
				- 1
				- x
			)

			var distance_top := y - wall_thickness

			var distance_bottom := (
				room_height
				- wall_thickness
				- 1
				- y
			)


			var edge_distance := min(
				distance_left,
				distance_right,
				distance_top,
				distance_bottom
			)


			# 只让草长在外围
			if edge_distance >= grass_depth:
				continue


			# ---------------------------------------------
			# 中央区域不长草
			# ---------------------------------------------

			var distance_to_center := Vector2(
				x,
				y
			).distance_to(center)


			if distance_to_center < center_clear_radius:
				continue


			# ---------------------------------------------
			# 越靠近墙草越密
			# ---------------------------------------------

			var edge_factor := 1.0 - (
				float(edge_distance)
				/ float(grass_depth)
			)


			var probability := (
				grass_density
				* edge_factor
			)


			if rng.randf() > probability:
				continue


			var tile := grass_tiles[
				rng.randi_range(
					0,
					grass_tiles.size() - 1
				)
			]


			grass.set_cell(
				cell,
				tile_source_id,
				tile
			)


# =========================================================
# Spawn / Door Positions
# =========================================================

func setup_spawn_points() -> void:

	var tile_size := 64.0


	var room_center := Vector2(
		room_width * tile_size * 0.5,
		room_height * tile_size * 0.5
	)


	# 玩家出生在中心稍微偏下
	player_spawn.position = (
		room_center
		+ Vector2(0, 96)
	)


	# 南入口
	entrance_south.position = Vector2(
		room_center.x,
		(room_height - wall_thickness) * tile_size
	)


	# 北出口
	exit_north.position = Vector2(
		room_center.x,
		wall_thickness * tile_size
	)


	# 中心灯
	center_light.position = room_center


# =========================================================
# Lighting
# =========================================================

func setup_light() -> void:

	center_light.energy = center_light_energy

	center_light.texture_scale = center_light_scale
```

```text

████████
████████
████████

中间：
▓▓▓▓▓▓▓▓

外围：
▒▒▒▒▒▒▒▒

透明
```

```text
白色

浅灰
```

# 三、核心脚本 room_base.gd

代码：

```gdscript
extends Node2D

class_name RoomBase


# ============================================================
# 《噬铁者》统一房间基类
#
# 负责：
# 1. 玩家进入检测
# 2. 战斗开始
# 3. 门锁定
# 4. 敌人数量管理
# 5. 清怪判定
# 6. 房间完成
# 7. 出口请求
# ============================================================


signal room_entered(room: RoomBase)

signal encounter_started(room: RoomBase)

signal room_cleared(room: RoomBase)

signal exit_requested(
	room: RoomBase,
	door: RoomDoor
)


# ============================================================
# Room Type
# ============================================================

enum RoomType {

	START,

	NORMAL,

	ELITE,

	SPECIAL,

	BOSS
}


@export_category("Room")

@export var room_id: StringName = &"ROOM_000"

@export var room_type: RoomType = RoomType.NORMAL


# 初始房等安全房直接视为完成
@export var starts_cleared: bool = false


# 战斗开始后是否锁门
@export var lock_doors_during_combat: bool = true


# ============================================================
# Node References
# ============================================================

@onready var ground: TileMapLayer = get_node_or_null("Ground")

@onready var grass: TileMapLayer = get_node_or_null("Grass")

@onready var walls: TileMapLayer = get_node_or_null("Walls")


@onready var enemy_container: Node2D = (
	get_node_or_null("EnemyContainer")
)


@onready var spawn_points: Node2D = (
	get_node_or_null("SpawnPoints")
)


@onready var doors_root: Node2D = (
	get_node_or_null("Doors")
)


@onready var player_detector: Area2D = (
	get_node_or_null("PlayerDetector")
)


@onready var player_spawn: Marker2D = (
	get_node_or_null("PlayerSpawn")
)


# ============================================================
# Runtime State
# ============================================================

var has_player_entered: bool = false

var encounter_active: bool = false

var cleared: bool = false


# 使用 Instance ID 管理敌人
var active_enemy_ids: Dictionary = {}


# ============================================================
# Ready
# ============================================================

func _ready() -> void:

	_connect_player_detector()

	_connect_doors()


	if starts_cleared or room_type == RoomType.START:

		cleared = true

		_set_all_doors_locked(false)

	else:

		cleared = false

		# 玩家没进入以前允许进入
		_set_all_doors_locked(false)


# ============================================================
# Player Detection
# ============================================================

func _connect_player_detector() -> void:

	if player_detector == null:
		return


	if not player_detector.body_entered.is_connected(
		_on_player_detector_body_entered
	):

		player_detector.body_entered.connect(
			_on_player_detector_body_entered
		)


func _on_player_detector_body_entered(
	body: Node2D
) -> void:

	if not body.is_in_group("player"):
		return


	if has_player_entered:
		return


	has_player_entered = true


	room_entered.emit(self)


	# 初始房 / 已清理房不进入战斗
	if cleared:
		return


	start_encounter()


# ============================================================
# Encounter
# ============================================================

func start_encounter() -> void:

	if encounter_active:
		return


	if cleared:
		return


	encounter_active = true


	if lock_doors_during_combat:

		_set_all_doors_locked(true)


	encounter_started.emit(self)


	# 子房间实现具体刷怪
	_spawn_room_enemies()


	# 注册已经生成的敌人
	_register_existing_enemies()


	# 如果没有怪，则直接完成
	if active_enemy_ids.is_empty():

		call_deferred("finish_room")


# ============================================================
# 子类覆盖
# ============================================================

func _spawn_room_enemies() -> void:

	pass


# ============================================================
# Enemy Register
# ============================================================

func register_enemy(enemy: Node) -> void:

	if enemy == null:
		return


	var enemy_id := enemy.get_instance_id()


	if active_enemy_ids.has(enemy_id):
		return


	active_enemy_ids[enemy_id] = true


	enemy.tree_exited.connect(
		_on_enemy_tree_exited.bind(enemy_id),
		CONNECT_ONE_SHOT
	)


# ============================================================

func _register_existing_enemies() -> void:

	if enemy_container == null:
		return


	for enemy in enemy_container.get_children():

		register_enemy(enemy)


# ============================================================

func _on_enemy_tree_exited(
	enemy_id: int
) -> void:

	active_enemy_ids.erase(enemy_id)


	if (
		encounter_active
		and active_enemy_ids.is_empty()
	):

		call_deferred("finish_room")


# ============================================================
# Room Complete
# ============================================================

func finish_room() -> void:

	if cleared:
		return


	cleared = true

	encounter_active = false


	_set_all_doors_locked(false)


	room_cleared.emit(self)


	_on_room_cleared()


# ============================================================
# 子类可覆盖
# ============================================================

func _on_room_cleared() -> void:

	pass


# ============================================================
# Doors
# ============================================================

func _connect_doors() -> void:

	if doors_root == null:
		return


	for node in doors_root.get_children():

		if node is RoomDoor:

			var door := node as RoomDoor


			if not door.used.is_connected(
				_on_door_used
			):

				door.used.connect(
					_on_door_used
				)


# ============================================================

func _set_all_doors_locked(
	value: bool
) -> void:

	if doors_root == null:
		return


	for node in doors_root.get_children():

		if node is RoomDoor:

			node.set_locked(value)


# ============================================================

func _on_door_used(
	door: RoomDoor
) -> void:

	if not cleared:
		return


	exit_requested.emit(
		self,
		door
	)


# ============================================================
# Spawn
# ============================================================

func get_player_spawn_position() -> Vector2:

	if player_spawn:

		return player_spawn.global_position


	return global_position
```

```gdscript
# 第一层：监牢入口

#

# 40 x 24 Tile
#

# 2 Tile
#

# 36 x 20 Tile
#

	T01_OPEN_ARENA,

	T02_HORIZONTAL_CORRIDOR,

	T03_VERTICAL_CORRIDOR,

	T04_L_SHAPE,

	T06_DOUBLE_CHAMBER,

	T09_PRISON_BARS,

	T10_BROKEN_ROOM

# ============================================================

# ============================================================
# Base Floor
# ============================================================

	for x in range(room_width):

		for y in range(room_height):

			ground.set_cell(

				Vector2i(x, y),

				tile_source_id,

				floor_tile

	var center_x: int = room_width / 2

	var center_y: int = room_height / 2

	var half_door: int = door_width / 2

	if north_door:

	if west_door:

			for x in range(wall_thickness):

				walls.erase_cell(

	# EAST

			center_y - half_door,

			for x in range(
				room_width - wall_thickness,

# ============================================================
# Grass / Moss
# ============================================================

func generate_grass() -> void:

	if grass_tiles.is_empty():

		return


	for x in range(
		wall_thickness,
		room_width - wall_thickness
	):

		for y in range(
			wall_thickness,
			room_height - wall_thickness
		):

			var cell := Vector2i(x, y)


			# 墙体上不生成

			if walls.get_cell_source_id(cell) != -1:

				continue


			var left_distance: int = (
				x - wall_thickness
			)

			var right_distance: int = (
				room_width
				- wall_thickness
				- 1
				- x
			)

			var top_distance: int = (
				y - wall_thickness
			)

			var bottom_distance: int = (
				room_height
				- wall_thickness
				- 1
				- y
			)


			var edge_distance: int = min(
				left_distance,
				right_distance,
				top_distance,
				bottom_distance
			)


			if edge_distance >= grass_edge_depth:

				continue


			var edge_factor: float = (

				1.0
				- float(edge_distance)
				/ float(grass_edge_depth)
			)


			var probability: float = (

				grass_density
				* edge_factor
			)


			if rng.randf() > probability:

				continue


			var tile_index := rng.randi_range(
				0,
				grass_tiles.size() - 1
			)


			grass.set_cell(

				cell,

				tile_source_id,

				grass_tiles[tile_index]
			)

# ============================================================

func _spawn_room_enemies() -> void:

	if enemy_pool.is_empty():

		return


	if spawn_points == null:

		return


	var points: Array[Marker2D] = []


	for child in spawn_points.get_children():

		if child is Marker2D:

			if child.name.begins_with(
				"AutoSpawn"
			):

				points.append(
					child
				)


	if points.is_empty():

		return


	var target_count := rng.randi_range(

		min_enemies,

		max_enemies
	)


	target_count = min(

		target_count,

		points.size()
	)


	# 随机选择出生点

	for i in range(target_count):

		var random_index := rng.randi_range(
			0,
			points.size() - 1
		)


		var spawn_point := points[
			random_index
		]


		points.remove_at(
			random_index
		)


		var enemy_scene := enemy_pool[
			rng.randi_range(
				0,
				enemy_pool.size() - 1
			)
		]


		spawn_enemy(
			enemy_scene,
			spawn_point
```

# 一、玩家固定生成在初始房中心

不要再通过：

```gdscript
room_width * 64 / 2
room_height * 64 / 2
```

决定玩家最终出生位置。

初始房场景：

```text
StartRoom_01
├── Ground
├── Grass
├── Walls
├── Decorations
├── CenterLight
├── Doors
├── PlayerDetector
└── PlayerSpawn
```

直接在 Godot 编辑器里把：

```text
PlayerSpawn (Marker2D)
```

拖到**房间中央**。

比如你的房间：

```text
40 × 24 Tile
Tile = 64 × 64
```

理论中心就是：

```text
X = 40 × 64 ÷ 2 = 1280
Y = 24 × 64 ÷ 2 = 768
```

所以 `PlayerSpawn.position` 可以是：

```gdscript
Vector2(1280, 768)
```

但更推荐**直接在编辑器里摆 Marker2D**，代码不要计算。

把之前 `StartRoom01.gd` 里的：

```gdscript
player_spawn.position = room_center + Vector2(0, 96)
```

删掉。

改成：

```gdscript
func setup_spawn_points() -> void:
	# PlayerSpawn 手动放在房间中央
	# 不再用代码改变 PlayerSpawn 的位置

	center_light.position = player_spawn.position
```

# 二、真正生成玩家时必须读取这个 Marker2D

这是最关键的部分。

假设你有：

```text
Main
├── Dungeon
└── Player
```

或者由 `DungeonGenerator.gd` 创建房间。

应该这样：

```gdscript
func spawn_player_in_start_room(
	player: Node2D,
	start_room: RoomBase
) -> void:

	if player == null:
		push_error("Player 不存在")
		return

	if start_room == null:
		push_error("StartRoom 不存在")
		return

	var spawn := start_room.get_node_or_null("PlayerSpawn")

	if spawn == null:
		push_error("StartRoom 缺少 PlayerSpawn")
		return

	player.global_position = spawn.global_position
```

然后地图初始化流程一定按照：

```text
生成初始房
↓
把初始房 add_child()
↓
生成 Player
↓
读取 PlayerSpawn.global_position
↓
设置 Player.global_position
```

例如：

```gdscript
@export var start_room_scene: PackedScene
@export var player_scene: PackedScene


func _ready() -> void:
	create_game()


func create_game() -> void:

	# 1. 创建初始房
	var start_room := start_room_scene.instantiate() as RoomBase

	add_child(start_room)

	start_room.position = Vector2.ZERO


	# 2. 创建玩家
	var player := player_scene.instantiate() as Node2D

	add_child(player)


	# 3. 玩家固定放到初始房中心
	var spawn := start_room.get_node(
		"PlayerSpawn"
	) as Marker2D

	player.global_position = spawn.global_position
```

这样无论以后初始房在：

```text
(0,0)
```

还是：

```text
(5000,3000)
```

玩家都会生成在正确位置。

### 不要这样写

```gdscript
player.position = Vector2(1280, 768)
```

因为这是**玩家相对于其父节点的局部坐标**。

推荐始终：

```gdscript
player.global_position = spawn.global_position
```

这样不容易因为 Dungeon、Room 有位移而出错。

## User 9
我使用的是64*64的瓦片，这会导致游戏地图生成经常出现问题，我该怎么解决

## GPT 12
64×64 本身并不会让 Godot 随机地图“不稳定”，真正容易出错的是**代码里同时混用了 Tile 坐标和像素坐标**。你现在前面的房间代码里确实已经出现了这种风险：`set_cell()` 用格子坐标，但玩家、房间实例和 Marker 又用了 `×64` 后的像素坐标。把坐标体系统一后，64×64 完全可以继续用。

## GPT 13
你现在最应该做的不是把 64×64 改成 32×32，而是**彻底取消代码里散落的 `64`**。让 TileSet 自己告诉代码瓦片尺寸，并且所有“格子→实际位置”的转换统一交给 `TileMapLayer.map_to_local()`；Godot 官方这个函数返回的就是某个格子的中心位置。

## GPT 14
会，而且你现在这些“偶发错位”很可能不是随机生成算法本身坏了，而是 **64×64 把半格偏移、局部坐标/全局坐标混用的问题放大了**。

最稳妥的做法是：**继续保留 64×64，不改素材尺寸；把整个地牢系统统一成“格子坐标驱动，像素坐标只在最后一步转换”。**

Godot 的 `TileSet.tile_size` 本身就保存了瓦片尺寸，而 `TileMapLayer.map_to_local()` / `local_to_map()` 就是官方提供的格子坐标与本地坐标转换接口。citeturn914418search0turn914418search1

## 第一件事：删除代码里所有写死的 `64`

你之前有很多这种：

```gdscript
var tile_size := 64.0

marker.position = Vector2(
	(cell.x + 0.5) * tile_size,
	(cell.y + 0.5) * tile_size
)
```

全部不要了。

改成统一函数：

```gdscript
func cell_to_global(cell: Vector2i) -> Vector2:

	return ground.to_global(
		ground.map_to_local(cell)
	)


func global_to_cell(global_pos: Vector2) -> Vector2i:

	var local_pos := ground.to_local(global_pos)

	return ground.local_to_map(local_pos)
```

因为 `map_to_local()` 返回的就是这个 Tile **中心点的本地坐标**，所以你根本不用自己 `+0.5 × 64`。citeturn914418search0

然后敌人出生点直接：

```gdscript
marker.global_position = cell_to_global(cell)
```

这样哪怕以后瓦片改成：

```text
32×32
48×48
64×64
96×96
```

这一段都不用改。

---

# 第二件事：确认 TileSet 真的是 64×64

Godot 里至少有两个尺寸概念需要一致。

你的 TileSet：

```text
Tile Size
64 × 64
```

Atlas Source：

```text
Texture Region Size
64 × 64
```

官方文档中 `TileSet.tile_size` 是地图的瓦片尺寸，而 `TileSetAtlasSource.texture_region_size` 是从图集中切出每块图像时使用的基础区域尺寸。citeturn914418search1turn561023search0

所以首先检查：

```text
TileSet
└── Tile Size = 64 × 64
```

然后：

```text
Atlas Source
└── Texture Region Size = 64 × 64
```

如果你的图片是 64×64，但 TileSet 还是：

```text
16×16
```

或者：

```text
32×32
```

就非常容易出现：

- Tile 重叠
- 碰撞位置不对
- 房间偏移
- 门对不齐
- Marker 和地图错位

---

# 第三个问题其实非常关键：40×24 是偶数尺寸

你之前的普通房间是：

```text
40 × 24 Tile
```

使用 64×64 后，实际大小是：

```text
2560 × 1536 px
```

这里有一个很容易忽略的问题：

**40×24 没有真正的“中心格”。**

例如宽度：

```text
0 1 2 ... 19 | 20 21 ... 39
```

真正几何中心在：

```text
19 与 20 之间
```

也就是：

```text
1280 px
```

但如果你写：

```gdscript
var center_x := room_width / 2
```

得到：

```text
20
```

然后认为：

```text
cell 20 = 房间中心
```

其实 cell 20 的中心是：

```text
20 × 64 + 32
= 1312
```

所以就出现了：

```text
1312 - 1280 = 32 px
```

**刚好偏半个 64×64 Tile。**

这就是为什么你会感觉地图“老是差一点”。

---

# 我建议你二选一

### 方案 A，我更推荐

把标准房间改成：

```text
41 × 25 Tile
```

这样：

```text
中心格 = Vector2i(20, 12)
```

而且可以完美做：

```text
3 Tile 宽的门
```

结构就是：

```text
0 ........ 19 [20] 21 ........ 40
                  ↑
               真中心
```

这样：

```gdscript
var center_cell := Vector2i(
	room_width / 2,
	room_height / 2
)
```

就是准确的。

对你现在这种房间制 ARPG，我比较推荐：

```text
普通房统一：41 × 25
门宽：3 Tile
墙厚：2 Tile
```

---

### 方案 B

继续使用：

```text
40 × 24
```

但是门宽改成：

```text
4 Tile
```

并且不要这么算：

```gdscript
var center_x := room_width / 2

for x in range(
	center_x - 1,
	center_x + 2
):
```

而应该：

```gdscript
var door_start_x := (
	room_width - door_width
) / 2


for x in range(
	door_start_x,
	door_start_x + door_width
):
	pass
```

例如：

```text
room_width = 40
door_width = 4
```

得到：

```text
(40 - 4) / 2
= 18
```

门：

```text
18
19
20
21
```

就正好关于房间中心对称。

# 第四件事：初始房和普通房不要使用不同“模块尺寸”

之前我们设计过：

```text
初始房：
30 × 18

普通房：
40 × 24
```

如果现在马上开始做随机房间拼接，我建议**不要这么做**。

第一版随机地牢全部统一成：

```text
房间槽位
41 × 25 Tile
```

例如：

```text
Start Room
41 × 25

Normal Room
41 × 25

Elite Room
41 × 25
```

视觉上初始房当然可以只有：

```text
30 × 18
```

那么让外围部分：

```text
空白 / 墙体 / 装饰
```

即可。

但是整个 `.tscn` 所占的**逻辑房间槽位**仍然保持：

```text
41 × 25
```

否则 DungeonGenerator 就必须处理：

```text
这个房宽 30
那个房宽 40
这个高 18
那个高 24
```

门连接、占位检测和防重叠都会明显复杂很多。

---

# 第五件事：DungeonGenerator 不要直接处理像素

这是解决随机地图问题最重要的一层。

不要存：

```text
房间 A = Vector2(2560, 0)

房间 B = Vector2(5120, 0)
```

而应该存：

```text
房间 A = Vector2i(0, 0)

房间 B = Vector2i(1, 0)

房间 C = Vector2i(1, -1)
```

这叫：

```text
Room Coordinate
```

例如：

```text
         (0,-1)
            │
(-1,0) ── (0,0) ── (1,0)
            │
          (0,1)
```

DungeonGenerator 逻辑里**只操作这个坐标**。

最后实例化场景的时候才转换：

```gdscript
func room_coord_to_world(
	room_coord: Vector2i
) -> Vector2:

	var tile_size := dungeon_tile_set.tile_size

	var room_pixel_size := Vector2(
		room_width * tile_size.x,
		room_height * tile_size.y
	)

	return Vector2(
		room_coord.x * room_pixel_size.x,
		room_coord.y * room_pixel_size.y
	)
```

假设：

```text
41 × 25
64 × 64
```

一个房间槽：

```text
宽：
41 × 64 = 2624 px

高：
25 × 64 = 1600 px
```

那么：

```text
Room (0,0)
→ (0,0)

Room (1,0)
→ (2624,0)

Room (-1,0)
→ (-2624,0)

Room (0,-1)
→ (0,-1600)
```

但是**地图算法本身完全不需要知道 2624 和 1600**。

它只知道：

```text
(0,0)
(1,0)
(-1,0)
```

这就稳定很多。

---

# 第六件事：房间内所有对象都尽量使用 Cell

例如敌人位置不要保存：

```gdscript
Vector2(832, 704)
```

而保存：

```gdscript
Vector2i(12, 10)
```

比如：

```gdscript
var enemy_spawn_cells: Array[Vector2i] = [
	Vector2i(10, 7),
	Vector2i(30, 7),
	Vector2i(10, 17),
	Vector2i(30, 17)
]
```

生成时：

```gdscript
for cell in enemy_spawn_cells:

	var marker := Marker2D.new()

	spawn_points.add_child(marker)

	marker.global_position = cell_to_global(cell)
```

玩家也是。

宝箱也是。

门也是。

柱子也是。

你最终应该形成这种规则：

```text
地图设计阶段
↓
全部使用 Vector2i Cell

需要显示/生成实体时
↓
map_to_local()

需要判断角色在哪格时
↓
local_to_map()
```

这正是 `TileMapLayer` 提供这两个坐标转换函数的用途。citeturn914418search0

# 第七件事：初始房中心不要再写 1280、768

可以直接计算真实房间几何中心。

例如：

```gdscript
func get_room_center_global() -> Vector2:

	var top_left := ground.map_to_local(
		Vector2i(0, 0)
	)

	var bottom_right := ground.map_to_local(
		Vector2i(
			room_width - 1,
			room_height - 1
		)
	)

	var center_local := (
		top_left + bottom_right
	) * 0.5

	return ground.to_global(
		center_local
	)
```

然后：

```gdscript
player.global_position = (
	start_room.get_room_center_global()
)
```

这样即使你坚持：

```text
40 × 24
```

角色也会生成在**真正的几何中心**，而不是偏 32 px。

---

# 第八件事：统一修改你之前的 `generate_spawn_points()`

之前：

```gdscript
var tile_size := Vector2(64, 64)

marker.position = Vector2(
	(cell.x + 0.5) * tile_size.x,
	(cell.y + 0.5) * tile_size.y
)
```

现在整个删掉。

改成：

```gdscript
func generate_spawn_points() -> void:

	if spawn_points == null:
		return


	for child in spawn_points.get_children():

		if child.name.begins_with(
			"AutoSpawn"
		):

			child.queue_free()


	for i in range(
		enemy_spawn_cells.size()
	):

		var marker := Marker2D.new()


		marker.name = (
			"AutoSpawn%02d"
			% (i + 1)
		)


		spawn_points.add_child(
			marker
		)


		var cell := (
			enemy_spawn_cells[i]
		)


		marker.global_position = (
			cell_to_global(cell)
		)
```

辅助函数：

```gdscript
func cell_to_global(
	cell: Vector2i
) -> Vector2:

	return ground.to_global(
		ground.map_to_local(cell)
	)
```

这一改非常重要。

---

# 第九件事：门也全部改成 Cell 定位

例如：

```gdscript
func get_north_door_cell() -> Vector2i:

	return Vector2i(
		room_width / 2,
		0
	)


func get_south_door_cell() -> Vector2i:

	return Vector2i(
		room_width / 2,
		room_height - 1
	)


func get_west_door_cell() -> Vector2i:

	return Vector2i(

		room_height / 2
	)


func get_east_door_cell() -> Vector2i:

	return Vector2i(
		room_width - 1,
		room_height / 2
	)
```

然后 Door 实体：

```gdscript
north_door.global_position = (
	cell_to_global(
		get_north_door_cell()
	)
)
```

不要再计算：

```text
2560 / 2
1536 / 2
```

检查：

```text
Room
Scale = (1,1)

Ground
Scale = (1,1)

Grass
Scale = (1,1)

Walls
Scale = (1,1)
```

尽量不要为了觉得：

> 64×64 看起来太大

## GPT 15
可以，这个功能很适合你现在这套“房间坐标驱动”的地牢结构。

你要的小地图核心需求其实是 4 件事：

1. **固定方向**：上北下南左西右东，不跟随玩家旋转  
2. **显示当前房间**
3. **显示与当前房间相连的房间**
4. **显示已经探索过的房间形状**

---

# 一、推荐的实现思路

不要去截主地图做缩略图。  
最稳的做法是：**单独画一个抽象拓扑小地图**。

也就是小地图不画真实 Tile，而是画：

- 房间块
- 房间之间的连接线 / 门
- 当前房间高亮
- 已探索房间常亮
- 当前相邻房间半高亮

这样最简单，也最稳定。

---

# 二、你现在需要的数据结构

你的 `DungeonGenerator` 最好维护这些数据：

```gdscript
var rooms_by_coord: Dictionary = {}
# key: Vector2i
# value: RoomBase

var explored_rooms: Dictionary = {}
# key: Vector2i
# value: true

var current_room_coord: Vector2i = Vector2i.ZERO
```

另外每个房间要知道自己的逻辑坐标：

```gdscript
# RoomBase.gd
var room_coord: Vector2i = Vector2i.ZERO
```


## 2）给普通房模板增加“获取开门方向”的函数

```gdscript
func get_open_directions() -> Dictionary:
	return {
		"north": north_door,
		"south": south_door,
		"east": east_door,
		"west": west_door
```

```gdscript
func get_open_directions() -> Dictionary:

		"south": false,
		"east": false,
		"west": false
	}
```

如果以后不同房间门不一样，这个函数仍然可以统一调用。

# 四、让 DungeonGenerator 记录房间并更新探索状态

下面给你一个**可直接用的逻辑骨架**。

## DungeonGenerator.gd

```gdscript
extends Node2D

class_name DungeonGenerator

signal minimap_updated


@export var start_room_scene: PackedScene
@export var normal_room_scenes: Array[PackedScene] = []
@export var player_scene: PackedScene


var rooms_by_coord: Dictionary = {}
var explored_rooms: Dictionary = {}
var current_room_coord: Vector2i = Vector2i.ZERO

var player: Node2D


const ROOM_GRID_SIZE := Vector2i(41, 25)  # 逻辑房间槽位
const TILE_SIZE := Vector2i(64, 64)       # 你的瓦片尺寸


func _ready() -> void:
	generate_test_dungeon()
	spawn_player()
	emit_signal("minimap_updated")

# 示例：先生成一个很简单的测试地牢
# ============================================================

func generate_test_dungeon() -> void:
	# 初始房
	var start_room := start_room_scene.instantiate() as RoomBase
	add_room(start_room, Vector2i(0, 0))

	# 周围几个房间
	if normal_room_scenes.size() > 0:
		var room1 := normal_room_scenes[0].instantiate() as RoomBase
		add_room(room1, Vector2i(0, -1))  # 北

	if normal_room_scenes.size() > 1:
		var room2 := normal_room_scenes[1].instantiate() as RoomBase
		add_room(room2, Vector2i(1, 0))   # 东

	if normal_room_scenes.size() > 2:
		var room3 := normal_room_scenes[2].instantiate() as RoomBase
		add_room(room3, Vector2i(1, -1))  # 东北

	if normal_room_scenes.size() > 3:
		var room4 := normal_room_scenes[3].instantiate() as RoomBase
		add_room(room4, Vector2i(0, 1))   # 南


func add_room(room: RoomBase, coord: Vector2i) -> void:
	room.room_coord = coord
	room.position = room_coord_to_world(coord)

	add_child(room)

	rooms_by_coord[coord] = room

	if not room.room_entered.is_connected(_on_room_entered):
		room.room_entered.connect(_on_room_entered)

	if not room.exit_requested.is_connected(_on_room_exit_requested):
		room.exit_requested.connect(_on_room_exit_requested)


func room_coord_to_world(coord: Vector2i) -> Vector2:
	var room_pixel_size := Vector2(
		ROOM_GRID_SIZE.x * TILE_SIZE.x,
		ROOM_GRID_SIZE.y * TILE_SIZE.y
	)

	return Vector2(
		coord.x * room_pixel_size.x,
		coord.y * room_pixel_size.y
	)


func spawn_player() -> void:
	player = player_scene.instantiate() as Node2D
	add_child(player)

	var start_room := rooms_by_coord.get(Vector2i.ZERO) as RoomBase
	if start_room:
		player.global_position = start_room.get_player_spawn_position()
		current_room_coord = start_room.room_coord
		explored_rooms[current_room_coord] = true


func _on_room_entered(room: RoomBase) -> void:
	current_room_coord = room.room_coord
	explored_rooms[current_room_coord] = true
	minimap_updated.emit()


func _on_room_exit_requested(room: RoomBase, door: RoomDoor) -> void:
	# 这里只演示移动到相邻房间
	var target_coord := room.room_coord + direction_to_offset(door.direction)

	if not rooms_by_coord.has(target_coord):
		return

	var target_room := rooms_by_coord[target_coord] as RoomBase

	player.global_position = target_room.get_player_spawn_position()

	current_room_coord = target_coord
	explored_rooms[current_room_coord] = true
	minimap_updated.emit()


func direction_to_offset(direction: int) -> Vector2i:
	match direction:
		RoomDoor.Direction.NORTH:
			return Vector2i(0, -1)
		RoomDoor.Direction.SOUTH:
			return Vector2i(0, 1)
		RoomDoor.Direction.EAST:
			return Vector2i(1, 0)
		RoomDoor.Direction.WEST:
			return Vector2i(-1, 0)
		_:
			return Vector2i.ZERO


func is_room_explored(coord: Vector2i) -> bool:
	return explored_rooms.has(coord)


func get_current_room() -> RoomBase:
	return rooms_by_coord.get(current_room_coord, null)


func get_room(coord: Vector2i) -> RoomBase:
	return rooms_by_coord.get(coord, null)
```

## MinimapView.gd

```gdscript
extends Control

class_name MinimapView


@export var dungeon_generator_path: NodePath

@export var room_size: Vector2 = Vector2(22, 22)
@export var room_gap: float = 10.0
@export var connector_width: float = 6.0

@export var explored_color: Color = Color(0.75, 0.75, 0.75, 0.9)
@export var current_color: Color = Color(1.0, 0.85, 0.2, 1.0)
@export var adjacent_color: Color = Color(0.45, 0.85, 1.0, 0.95)
@export var unexplored_connected_color: Color = Color(0.35, 0.35, 0.35, 0.75)
@export var line_color: Color = Color(0.9, 0.9, 0.9, 0.8)
@export var background_grid_color: Color = Color(0.2, 0.2, 0.2, 0.2)

@export var visible_radius: int = 4
# 只画当前房间附近几格，避免整个地图太大


var dungeon: DungeonGenerator


func _ready() -> void:
	if dungeon_generator_path != NodePath():
		dungeon = get_node(dungeon_generator_path) as DungeonGenerator

	if dungeon:
		if not dungeon.minimap_updated.is_connected(queue_redraw):
			dungeon.minimap_updated.connect(queue_redraw)

	queue_redraw()


func _draw() -> void:
	if dungeon == null:
		return

	var center := size * 0.5
	var current_coord := dungeon.current_room_coord

	# 先画连接线，再画房间
	for coord in dungeon.rooms_by_coord.keys():
		if not _should_draw_room(coord, current_coord):
			continue

		var room := dungeon.get_room(coord)
		if room == null:
			continue

		var room_center := minimap_pos_from_coord(coord, current_coord, center)
		draw_connections(room, coord, current_coord, room_center)

	for coord in dungeon.rooms_by_coord.keys():
		if not _should_draw_room(coord, current_coord):
			continue

		var room := dungeon.get_room(coord)
		if room == null:
			continue

		var room_center := minimap_pos_from_coord(coord, current_coord, center)
		draw_room(room, coord, current_coord, room_center)


func _should_draw_room(coord: Vector2i, current_coord: Vector2i) -> bool:
	var delta := coord - current_coord

	if abs(delta.x) > visible_radius:
		return false
	if abs(delta.y) > visible_radius:
		return false

	# 已探索房间：显示
	if dungeon.is_room_explored(coord):
		return true

	# 未探索但与当前房间相连：显示
	if is_adjacent_to_current(coord, current_coord):
		return true

	return false


func is_adjacent_to_current(coord: Vector2i, current_coord: Vector2i) -> bool:
	var delta := coord - current_coord
	var adjacent_offsets := [
		Vector2i(0, -1),
		Vector2i(0, 1),
		Vector2i(1, 0),
		Vector2i(-1, 0)
	]

	for offset in adjacent_offsets:
		if delta == offset:
			return true

	return false


func minimap_pos_from_coord(coord: Vector2i, current_coord: Vector2i, center: Vector2) -> Vector2:
	var delta := coord - current_coord

	var step := room_size + Vector2(room_gap, room_gap)

	return center + Vector2(
		delta.x * step.x,
		delta.y * step.y
	)


func draw_room(room: RoomBase, coord: Vector2i, current_coord: Vector2i, room_center: Vector2) -> void:
	var rect := Rect2(
		room_center - room_size * 0.5,
		room_size
	)

	var color := explored_color

	if coord == current_coord:
		color = current_color
	elif is_adjacent_to_current(coord, current_coord) and not dungeon.is_room_explored(coord):
		color = adjacent_color
	elif not dungeon.is_room_explored(coord):
		color = unexplored_connected_color

	draw_rect(rect, color, true)
	draw_rect(rect, Color.BLACK, false, 2.0)

	# 画门的方向缺口 / 短线
	draw_room_exits(room, rect, color)


func draw_room_exits(room: RoomBase, rect: Rect2, color: Color) -> void:
	if not room.has_method("get_open_directions"):
		return

	var exits: Dictionary = room.get_open_directions()

	var cx := rect.get_center().x
	var cy := rect.get_center().y

	var half_w := rect.size.x * 0.5
	var half_h := rect.size.y * 0.5
	var stub := 6.0

	if exits.get("north", false):
		draw_line(
			Vector2(cx, cy - half_h),
			Vector2(cx, cy - half_h - stub),
			line_color,
			2.0
		)

	if exits.get("south", false):
		draw_line(
			Vector2(cx, cy + half_h),
			Vector2(cx, cy + half_h + stub),
			line_color,
			2.0
		)

	if exits.get("west", false):
		draw_line(
			Vector2(cx - half_w, cy),
			Vector2(cx - half_w - stub, cy),
			line_color,
			2.0
		)

	if exits.get("east", false):
		draw_line(
			Vector2(cx + half_w, cy),
			Vector2(cx + half_w + stub, cy),
			line_color,
			2.0
		)


func draw_connections(room: RoomBase, coord: Vector2i, current_coord: Vector2i, room_center: Vector2) -> void:
	if not room.has_method("get_open_directions"):
		return

	var exits: Dictionary = room.get_open_directions()

	if exits.get("north", false):
		draw_connection_if_visible(coord, Vector2i(0, -1), current_coord, room_center)

	if exits.get("south", false):
		draw_connection_if_visible(coord, Vector2i(0, 1), current_coord, room_center)

	if exits.get("east", false):
		draw_connection_if_visible(coord, Vector2i(1, 0), current_coord, room_center)

	if exits.get("west", false):
		draw_connection_if_visible(coord, Vector2i(-1, 0), current_coord, room_center)


func draw_connection_if_visible(coord: Vector2i, offset: Vector2i, current_coord: Vector2i, room_center: Vector2) -> void:
	var target_coord := coord + offset

	if not dungeon.rooms_by_coord.has(target_coord):
		return

	if not _should_draw_room(target_coord, current_coord):
		return

	var center := size * 0.5
	var target_center := minimap_pos_from_coord(target_coord, current_coord, center)

	draw_line(room_center, target_center, line_color, connector_width)
```

# 十、方向固定是怎么保证的

因为我们始终使用房间逻辑坐标：

```gdscript
Vector2i(0, -1)  # 北
Vector2i(0, 1)   # 南
Vector2i(1, 0)   # 东
Vector2i(-1, 0)  # 西
```

并且在小地图里：

```gdscript
delta.y * step.y
```

是直接向下绘制的，所以：

- 上方永远是北
- 下方永远是南
- 左边永远是西
- 右边永远是东

即使玩家面朝别的方向，地图也不会旋转。

# 三、RoomBase.gd

这是所有房间的基类。

## room_base.gd

```gdscript
extends Node2D

class_name RoomBase


signal room_entered(room: RoomBase)


enum RoomType {
	START,
	NORMAL,
	ELITE,
	SPECIAL,
	BOSS
}


@export_category("Room")

@export var room_id: StringName = &"ROOM"

@export var room_type: RoomType = RoomType.NORMAL


# ============================================================
# 房间逻辑坐标
#
# 例如：
#
# (0, -1)
#    │
# (0, 0) ── (1, 0)
#
# ============================================================

var room_coord: Vector2i = Vector2i.ZERO


# ============================================================
# 房间尺寸
# ============================================================

@export_category("Room Size")

@export var room_size_cells: Vector2i = Vector2i(
	41,
	25
)

@export_range(1, 5)
var wall_thickness: int = 2


# ============================================================
# Nodes
# ============================================================

@onready var ground: TileMapLayer = (
	get_node_or_null("Ground") as TileMapLayer
)

@onready var grass: TileMapLayer = (
	get_node_or_null("Grass") as TileMapLayer
)

@onready var walls: TileMapLayer = (
	get_node_or_null("Walls") as TileMapLayer
)

@onready var decorations: Node2D = (
	get_node_or_null("Decorations") as Node2D
)

@onready var enemy_container: Node2D = (
	get_node_or_null("EnemyContainer") as Node2D
)

@onready var spawn_points: Node2D = (
	get_node_or_null("SpawnPoints") as Node2D
)

@onready var player_detector: Area2D = (
	get_node_or_null("PlayerDetector") as Area2D
)

@onready var player_spawn: Marker2D = (
	get_node_or_null("PlayerSpawn") as Marker2D
)


func _ready() -> void:

	_validate_room()

	_setup_render_layers()

	_setup_tile_collisions()

	_setup_player_spawn()

	_setup_player_detector()


# ============================================================
# 检查
# ============================================================

func _validate_room() -> void:

	if ground == null:
		push_error(
			"%s 缺少 Ground TileMapLayer"
			% name
		)
		return


	if ground.tile_set == null:
		push_error(
			"%s 的 Ground 没有 TileSet"
			% name
		)


# ============================================================
# 绘制层级
# ============================================================

func _setup_render_layers() -> void:

	if ground:
		ground.z_index = 0


	if grass:
		grass.z_index = 1


	if walls:
		walls.z_index = 2


	if decorations:
		decorations.z_index = 3


# ============================================================
# 碰撞
#
# Ground / Grass 只负责显示。
#
# Walls 才参与碰撞。
# ============================================================

func _setup_tile_collisions() -> void:

	if ground:
		ground.collision_enabled = false


	if grass:
		grass.collision_enabled = false


	if walls:
		walls.collision_enabled = true


# ============================================================
# Tile Size
# ============================================================

func get_tile_size() -> Vector2i:

	if ground == null:
		return Vector2i.ZERO


	if ground.tile_set == null:
		return Vector2i.ZERO


	return ground.tile_set.tile_size


# ============================================================
# 房间实际像素大小
#
# 不写死 64。
#
# 如果 TileSet 是：
#
# 64 × 64
#
# 那么：
#
# 41 × 25
#
# 会自动得到：
#
# 2624 × 1600
# ============================================================

func get_room_pixel_size() -> Vector2:

	var tile_size := get_tile_size()


	if tile_size == Vector2i.ZERO:
		return Vector2.ZERO


	return Vector2(

		room_size_cells.x
			* tile_size.x,

		room_size_cells.y
			* tile_size.y
	)


# ============================================================
# 中心格
#
# 41 × 25：
#
# 中心 = (20, 12)
# ============================================================

func get_center_cell() -> Vector2i:

	return Vector2i(

		room_size_cells.x / 2,

		room_size_cells.y / 2
	)


# ============================================================
# Cell -> Global
#
# 所有实体位置尽量使用这个函数。
# ============================================================

func cell_to_global(
	cell: Vector2i
) -> Vector2:

	if ground == null:
		return global_position


	return ground.to_global(
		ground.map_to_local(cell)
	)


# ============================================================
# Global -> Cell
# ============================================================

func global_to_cell(
	world_position: Vector2
) -> Vector2i:

	if ground == null:
		return Vector2i.ZERO


	var local_position := ground.to_local(
		world_position
	)


	return ground.local_to_map(
		local_position
	)


# ============================================================
# 玩家出生点
#
# 永远固定到房间真正的中心格。
# ============================================================

func _setup_player_spawn() -> void:

	if player_spawn == null:
		return


	player_spawn.global_position = (
		cell_to_global(
			get_center_cell()
		)
	)


func get_player_spawn_position() -> Vector2:

	if player_spawn:
		return player_spawn.global_position


	return cell_to_global(
		get_center_cell()
	)


# ============================================================
# 房间检测区域
#
# Area2D 自动覆盖墙内区域。
# ============================================================

func _setup_player_detector() -> void:

	if player_detector == null:
		return


	var collision_shape := (
		player_detector.get_node_or_null(
			"CollisionShape2D"
		) as CollisionShape2D
	)


	if collision_shape == null:
		push_error(
			"%s 的 PlayerDetector 缺少 CollisionShape2D"
			% name
		)

		return


	var tile_size := get_tile_size()


	if tile_size == Vector2i.ZERO:
		return


	var rectangle := RectangleShape2D.new()


	var interior_width := (
		room_size_cells.x
		- wall_thickness * 2
	)


	var interior_height := (
		room_size_cells.y
		- wall_thickness * 2
	)


	rectangle.size = Vector2(

		interior_width
			* tile_size.x,

		interior_height
			* tile_size.y
	)


	collision_shape.shape = rectangle


	# 房间整体中心
	collision_shape.position = (
		get_room_pixel_size()
		* 0.5
	)


	player_detector.collision_layer = 0

	player_detector.collision_mask = 1

	player_detector.monitoring = true

	player_detector.monitorable = true


	if not player_detector.body_entered.is_connected(
		_on_player_detector_body_entered
	):

		player_detector.body_entered.connect(
			_on_player_detector_body_entered
		)


# ============================================================
# 玩家进入房间
# ============================================================

func _on_player_detector_body_entered(
	body: Node2D
) -> void:

	if not body.is_in_group("player"):
		return


	room_entered.emit(self)

# ============================================================

		"south": false,
		"east": false,
```

```gdscript
	if grass == null:
		return


	if grass_tiles.is_empty():
		return


	for x in range(
		wall_thickness,
		room_size_cells.x
			- wall_thickness
	):

		for y in range(
			wall_thickness,
			room_size_cells.y
				- wall_thickness
		):

			var cell := Vector2i(
				x,
				y
			)


			if walls.get_cell_source_id(
				cell
			) != -1:

				continue


			var distance_to_edge: int = min(

				x - wall_thickness,

				room_size_cells.x
					- wall_thickness
					- 1
					- x,

				y - wall_thickness,

				room_size_cells.y
					- wall_thickness
					- 1
					- y
			)


			if distance_to_edge >= (
				grass_edge_depth
			):

				continue


			var edge_factor := (

				1.0
				- float(
					distance_to_edge
				)
				/ float(
					grass_edge_depth
				)
			)


			if rng.randf() > (
				grass_density
				* edge_factor
			):

				continue


			var tile := grass_tiles[
				rng.randi_range(
					0,
					grass_tiles.size()
						- 1
				)
			]


			grass.set_cell(

				cell,

				tile_source_id,

				tile
			)
```

```text
1280
```

地图逻辑只认：

```text
Vector2i(0,0)
```

最后才转换成实际世界坐标。

## dungeon_generator.gd

```gdscript
extends Node2D

class_name DungeonGenerator


signal minimap_updated


# ============================================================
# Scenes
# ============================================================

@export_category("Scenes")

@export var room_scene: PackedScene

@export var player_scene: PackedScene


# ============================================================
# Dungeon
# ============================================================

@export_category("Dungeon")

@export_range(4, 30)
var room_count: int = 12


@export_range(2, 10)
var max_distance_from_start: int = 4


@export var generation_seed: int = 0


# ============================================================
# Runtime
# ============================================================

var rooms_by_coord: Dictionary = {}

var links_by_coord: Dictionary = {}

var explored_rooms: Dictionary = {}


var generated_coords: Array[Vector2i] = []


var current_room_coord: Vector2i = (
	Vector2i.ZERO
)


var player: CharacterBody2D


var rng := RandomNumberGenerator.new()


const NORTH := Vector2i(0, -1)

const SOUTH := Vector2i(0, 1)

const EAST := Vector2i(1, 0)

const WEST := Vector2i(-1, 0)


const CARDINAL_DIRECTIONS := [

	NORTH,

	SOUTH,

	EAST,

	WEST
]


# ============================================================
# Tree
# ============================================================

func _enter_tree() -> void:

	add_to_group(
		"dungeon_generator"
	)


func _ready() -> void:

	if generation_seed == 0:

		rng.randomize()

	else:

		rng.seed = generation_seed


	if room_scene == null:

		push_error(
			"DungeonGenerator 没有设置 room_scene"
		)

		return


	if player_scene == null:

		push_error(
			"DungeonGenerator 没有设置 player_scene"
		)

		return


	_generate_coordinates()

	_build_room_links()

	_spawn_rooms()

	_spawn_player()

	_mark_room_explored(
		Vector2i.ZERO
	)

	minimap_updated.emit()


# ============================================================
# Step 1
#
# 生成房间逻辑坐标。
#
# 使用随机游走保证整个地图天然连通。
# ============================================================

func _generate_coordinates() -> void:

	generated_coords.clear()


	var occupied: Dictionary = {}


	var current := Vector2i.ZERO


	generated_coords.append(
		current
	)

	occupied[current] = true


	var attempts := 0

	var max_attempts := (
		room_count * 200
	)


	while (
		generated_coords.size()
		< room_count
		and attempts
		< max_attempts
	):

		attempts += 1


		var direction: Vector2i = (
			CARDINAL_DIRECTIONS[
				rng.randi_range(
					0,
					CARDINAL_DIRECTIONS.size()
						- 1
				)
			]
		)


		var next := (
			current
			+ direction
		)


		if abs(next.x) > (
			max_distance_from_start
		):

			continue


		if abs(next.y) > (
			max_distance_from_start
		):

			continue


		current = next


		if occupied.has(
			current
		):

			continue


		occupied[current] = true


		generated_coords.append(
			current
		)


	if generated_coords.size() < (
		room_count
	):

		push_warning(
			"没有生成到目标房间数量：%d / %d"
			% [
				generated_coords.size(),
				room_count
			]
		)


# ============================================================
# Step 2
#
# 构造真实房间连接图。
#
# 只有互相正交相邻的生成房间才连接。
# ============================================================

func _build_room_links() -> void:

	links_by_coord.clear()


	var occupied: Dictionary = {}


	for coord in generated_coords:

		occupied[coord] = true


	for coord in generated_coords:

		var neighbors: Array[Vector2i] = []


		for direction in (
			CARDINAL_DIRECTIONS
		):

			var neighbor := (
				coord
				+ direction
			)


			if occupied.has(
				neighbor
			):

				neighbors.append(
					neighbor
				)


		links_by_coord[coord] = (
			neighbors
		)


# ============================================================
# Step 3
#
# 创建房间
# ============================================================

func _spawn_rooms() -> void:

	for coord in generated_coords:

		var room := (
			room_scene.instantiate()
			as Layer1RoomTemplate
		)


		if room == null:

			push_error(
				"room_scene 必须使用 Layer1RoomTemplate.gd"
			)

			return


		room.room_coord = coord


		# 初始房永远使用开阔房

		if coord == Vector2i.ZERO:

			room.room_type = (
				RoomBase.RoomType.START
			)

			room.template_type = (
				Layer1RoomTemplate
				.TemplateType
				.T01_OPEN_ARENA
			)

		else:

			room.room_type = (
				RoomBase.RoomType.NORMAL
			)


			room.template_type = (
				rng.randi_range(
					0,
					9
				)
			)


		room.generation_seed = (
			_get_room_seed(
				coord
			)
		)


		room.north_door = (
			are_coords_linked(
				coord,
				coord + NORTH
			)
		)


		room.south_door = (
			are_coords_linked(
				coord,
				coord + SOUTH
			)
		)


		room.east_door = (
			are_coords_linked(
				coord,
				coord + EAST
			)
		)


		room.west_door = (
			are_coords_linked(
				coord,
				coord + WEST
			)
		)


		add_child(
			room
		)


		var room_pixel_size := (
			room.get_room_pixel_size()
		)


		if room_pixel_size == (
			Vector2.ZERO
		):

			push_error(
				"房间 TileSet.tile_size 无效"
			)

			return


		room.position = Vector2(

			coord.x
				* room_pixel_size.x,

			coord.y
				* room_pixel_size.y
		)


		rooms_by_coord[coord] = (
			room
		)


		room.room_entered.connect(
			_on_room_entered
		)


# ============================================================
# 玩家
# ============================================================

func _spawn_player() -> void:

	player = (
		player_scene.instantiate()
		as CharacterBody2D
	)


	if player == null:

		push_error(
			"player_scene 根节点必须是 CharacterBody2D"
		)

		return


	add_child(
		player
	)


	var start_room := get_room(
		Vector2i.ZERO
	)


	if start_room == null:
		return


	# ========================================================
	# 玩家绝对固定生成在初始房中心
	# ========================================================

	player.global_position = (
		start_room
			.get_player_spawn_position()
	)


	current_room_coord = (
		Vector2i.ZERO
	)


# ============================================================
# Room Entered
# ============================================================

func _on_room_entered(
	room: RoomBase
) -> void:

	if room == null:
		return


	var new_coord := (
		room.room_coord
	)


	var room_changed := (
		new_coord
		!= current_room_coord
	)


	current_room_coord = (
		new_coord
	)


	var newly_explored := (
		not explored_rooms.has(
			new_coord
		)
	)


	_mark_room_explored(
		new_coord
	)


	if (
		room_changed
		or newly_explored
	):

		minimap_updated.emit()


# ============================================================
# Explored
# ============================================================

func _mark_room_explored(
	coord: Vector2i
) -> void:

	explored_rooms[coord] = true


func is_room_explored(
	coord: Vector2i
) -> bool:

	return explored_rooms.has(
		coord
	)


# ============================================================
# Room
# ============================================================

func get_room(
	coord: Vector2i
) -> RoomBase:

	return (
		rooms_by_coord.get(
			coord,
			null
		)
		as RoomBase
	)


# ============================================================
# Links
# ============================================================

func get_linked_neighbors(
	coord: Vector2i
) -> Array[Vector2i]:

	var result: Array[Vector2i] = []


	if not links_by_coord.has(
		coord
	):

		return result


	var raw_neighbors: Array = (
		links_by_coord[coord]
	)


	for neighbor in raw_neighbors:

		result.append(
			neighbor
		)


	return result


func are_coords_linked(
	first: Vector2i,
	second: Vector2i
) -> bool:

	if not links_by_coord.has(
		first
	):

		return false


	var neighbors: Array = (
		links_by_coord[first]
	)


	return neighbors.has(
		second
	)

# Minimap Visibility
#
# 显示：
#
# 已探索房间
#
# +
#
# 当前房间直接连接的未探索房间
# ============================================================

func should_show_room_on_minimap(
	coord: Vector2i
) -> bool:

	if is_room_explored(
		coord
	):

		return true


	if are_coords_linked(
		current_room_coord,
		coord
	):

		return true


	return false

func _get_room_seed(

	var base_seed := generation_seed


	if base_seed == 0:
		base_seed = 918273


	return (

		base_seed
		+ coord.x * 73856093
		+ coord.y * 19349663
	)
```

## minimap_view.gd

```gdscript
extends Control

class_name MinimapView


# ============================================================
# Layout
# ============================================================

@export_category("Layout")

@export var room_box_size := Vector2(
	28.0,
	20.0
)

@export var room_gap: float = 14.0

@export var margin: float = 18.0

@export var connector_width: float = 5.0


# ============================================================
# Colors
# ============================================================

@export_category("Colors")

@export var current_room_color := Color(
	1.0,
	0.82,
	0.16,
	1.0
)

@export var explored_room_color := Color(
	0.78,
	0.80,
	0.84,
	1.0
)

@export var adjacent_room_color := Color(
	0.30,
	0.67,
	0.95,
	0.85
)

@export var connection_color := Color(
	0.68,
	0.72,
	0.78,
	0.90
)

@export var outline_color := Color(
	0.08,
	0.08,
	0.10,
	1.0
)

@export var player_color := Color(
	1.0,
	1.0,
	1.0,
	1.0
)


var dungeon: DungeonGenerator


func _ready() -> void:

	mouse_filter = (
		Control.MOUSE_FILTER_IGNORE
	)


	_find_dungeon_generator()


	if dungeon:

		if not dungeon.minimap_updated.is_connected(
			_on_minimap_updated
		):

			dungeon.minimap_updated.connect(
				_on_minimap_updated
			)


	queue_redraw()


# ============================================================
# Find Dungeon
# ============================================================

func _find_dungeon_generator() -> void:

	var generators := (
		get_tree()
			.get_nodes_in_group(
				"dungeon_generator"
			)
	)


	if generators.is_empty():

		push_error(
			"Minimap 找不到 DungeonGenerator"
		)

		return


	dungeon = (
		generators[0]
		as DungeonGenerator
	)


# ============================================================
# Update
# ============================================================

func _on_minimap_updated() -> void:

	queue_redraw()


func _notification(
	what: int
) -> void:

	if what == (
		NOTIFICATION_RESIZED
	):

		queue_redraw()


# ============================================================
# Draw
# ============================================================

func _draw() -> void:

	if dungeon == null:
		return


	var visible_coords := (
		_get_visible_rooms()
	)


	if visible_coords.is_empty():
		return


	var bounds := (
		_get_bounds(
			visible_coords
		)
	)


	var scale_factor := (
		_calculate_scale(
			bounds
		)
	)


	var scaled_room_size := (
		room_box_size
		* scale_factor
	)


	var step := Vector2(

		(
			room_box_size.x
			+ room_gap
		)
		* scale_factor,

		(
			room_box_size.y
			+ room_gap
		)
		* scale_factor
	)


	var grid_center := Vector2(

		(
			bounds.position.x
			+ bounds.end.x
			- 1
		)
		* 0.5,

		(
			bounds.position.y
			+ bounds.end.y
			- 1
		)
		* 0.5
	)


	var screen_center := (
		size * 0.5
	)


	# ========================================================
	# 快速查询
	# ========================================================

	var visible_lookup: Dictionary = {}


	for coord in visible_coords:

		visible_lookup[coord] = true


	# ========================================================
	# 先画连接线
	# ========================================================

	for coord in visible_coords:

		var neighbors := (
			dungeon
				.get_linked_neighbors(
					coord
				)
		)


		for neighbor in neighbors:

			if not visible_lookup.has(
				neighbor
			):

				continue


			# 防止一条连接画两次

			if not _should_draw_connection(
				coord,
				neighbor
			):

				continue


			var from_position := (
				_coord_to_screen(
					coord,
					grid_center,
					screen_center,
					step
				)
			)


			var to_position := (
				_coord_to_screen(
					neighbor,
					grid_center,
					screen_center,
					step
				)
			)


			draw_line(

				from_position,

				to_position,

				connection_color,

				max(
					2.0,
					connector_width
						* scale_factor
				),

				true
			)


	# ========================================================
	# 再画房间
	# ========================================================

	for coord in visible_coords:

		var center := (
			_coord_to_screen(
				coord,
				grid_center,
				screen_center,
				step
			)
		)


		_draw_room(

			coord,

			center,

			scaled_room_size,

			scale_factor
		)


# ============================================================
# Visible Rooms
# ============================================================

func _get_visible_rooms() -> Array[Vector2i]:

	var result: Array[Vector2i] = []


	for key in (
		dungeon.rooms_by_coord.keys()
	):

		var coord: Vector2i = key


		if dungeon.should_show_room_on_minimap(
			coord
		):

			result.append(
				coord
			)


	return result


# ============================================================
# Bounds
# ============================================================

func _get_bounds(
	coords: Array[Vector2i]
) -> Rect2i:

	var min_x := coords[0].x

	var max_x := coords[0].x

	var min_y := coords[0].y

	var max_y := coords[0].y


	for coord in coords:

		min_x = min(
			min_x,
			coord.x
		)

		max_x = max(
			max_x,
			coord.x
		)

		min_y = min(
			min_y,
			coord.y
		)

		max_y = max(
			max_y,
			coord.y
		)


	return Rect2i(

		min_x,

		min_y,

		max_x
			- min_x
			+ 1,

		max_y
			- min_y
			+ 1
	)


# ============================================================
# 自动缩放
#
# 走过的房间越来越多时自动缩小地图。
#
# 不旋转。
# ============================================================

func _calculate_scale(
	bounds: Rect2i
) -> float:

	var width_rooms := (
		bounds.size.x
	)


	var height_rooms := (
		bounds.size.y
	)


	var raw_width := (

		room_box_size.x

		+ max(
			0,
			width_rooms - 1
		)

		* (
			room_box_size.x
			+ room_gap
		)
	)


	var raw_height := (

		room_box_size.y

		+ max(
			0,
			height_rooms - 1
		)

		* (
			room_box_size.y
			+ room_gap
		)
	)


	var available_width := max(

		1.0,

		size.x
			- margin * 2.0
	)


	var available_height := max(

		1.0,

		size.y
			- margin * 2.0
	)


	var scale_x := (
		available_width
		/ raw_width
	)


	var scale_y := (
		available_height
		/ raw_height
	)


	return clamp(

		min(
			scale_x,
			scale_y
		),

		0.35,

		1.0
	)


# ============================================================
# Coordinate -> Screen
#
# 注意：
#
# coord.y = -1
#
# 会出现在屏幕上方。
#
# 所以地图方向永远固定：
#
# 北 ↑
# 南 ↓
# 西 ←
# 东 →
# ============================================================

func _coord_to_screen(

	coord: Vector2i,

	grid_center: Vector2,

	screen_center: Vector2,

	step: Vector2

) -> Vector2:

	var delta := (

		Vector2(coord)

		- grid_center
	)


	return (

		screen_center

		+ Vector2(

			delta.x
				* step.x,

			delta.y
				* step.y
		)
	)


# ============================================================
# Room
# ============================================================

func _draw_room(

	coord: Vector2i,

	center: Vector2,

	scaled_room_size: Vector2,

	scale_factor: float

) -> void:

	var room_rect := Rect2(

		center
			- scaled_room_size
			* 0.5,

		scaled_room_size
	)


	var fill_color := (
		explored_room_color
	)


	# 当前房间

	if coord == (
		dungeon.current_room_coord
	):

		fill_color = (
			current_room_color
		)


	# 当前相邻但未探索

	elif not dungeon.is_room_explored(
		coord
	):

		fill_color = (
			adjacent_room_color
		)


	draw_rect(

		room_rect,

		fill_color,

		true
	)


	draw_rect(

		room_rect,

		outline_color,

		false,

		max(
			1.0,
			2.0 * scale_factor
		)
	)


	# ========================================================
	# 当前玩家标记
	# ========================================================

	if coord == (
		dungeon.current_room_coord
	):

		draw_circle(

			center,

			max(
				2.5,
				4.0
					* scale_factor
			),

			player_color
		)


# ============================================================
# 避免重复画线
# ============================================================

func _should_draw_connection(

	first: Vector2i,

	second: Vector2i

) -> bool:

	if first.x < second.x:
		return true


	if first.x > second.x:
		return false


	return first.y < second.y
```

下面按你上一版工程直接改，不需要重做 64×64 地图。门的碰撞开关我用 `set_deferred("disabled", ...)`，因为 Godot 当前文档明确要求 `CollisionShape2D.disabled` 通过 deferred 修改。citeturn300653search0

# 1. 新增 `room_door.gd`

```gdscript
extends StaticBody2D

class_name RoomDoor


enum Direction {
	NORTH,
	SOUTH,
	EAST,
	WEST
}


var direction: int = Direction.NORTH
var locked: bool = false


var collision_shape: CollisionShape2D
var visual: Polygon2D


func _ready() -> void:
	_ensure_nodes()


func configure(
	new_direction: int,
	blocker_size: Vector2
) -> void:

	direction = new_direction

	_ensure_nodes()


	var rectangle := RectangleShape2D.new()
	rectangle.size = blocker_size

	collision_shape.shape = rectangle


	var half := blocker_size * 0.5

	visual.polygon = PackedVector2Array([
		Vector2(-half.x, -half.y),
		Vector2(half.x, -half.y),
		Vector2(half.x, half.y),
		Vector2(-half.x, half.y)
	])


	# 临时战斗门颜色
	# 后面换成你的铁门 Sprite 即可
	visual.color = Color(
		0.34,
		0.25,
		0.18,
		0.95
	)


	set_locked(false)


func _ensure_nodes() -> void:

	if collision_shape == null:

		collision_shape = CollisionShape2D.new()
		collision_shape.name = "CollisionShape2D"

		add_child(collision_shape)


	if visual == null:

		visual = Polygon2D.new()
		visual.name = "DoorVisual"

		visual.z_index = 9

		add_child(visual)


func set_locked(value: bool) -> void:

	locked = value


	if collision_shape:

		# Godot 要求 deferred 修改物理碰撞状态
		collision_shape.set_deferred(
			"disabled",
			not locked
		)


	if visual:

		visual.visible = locked
```

它现在是一个**程序生成的临时门**：

```text
房间没战斗
→ invisible
→ collision disabled


进入战斗
→ visible
→ collision enabled


清怪
→ invisible
→ collision disabled
```

后面换铁门动画时不用动战斗逻辑。

# 2. 用下面版本替换 `RoomBase.gd`

这一步是核心。

```gdscript
extends Node2D

class_name RoomBase


signal room_entered(room: RoomBase)

signal encounter_started(room: RoomBase)

signal room_cleared(room: RoomBase)


enum RoomType {
	START,
	NORMAL,
	ELITE,

	BOSS

# ============================================================
# Room
# ============================================================

@export_category("Room")

@export var room_id: StringName = &"ROOM"

@export var room_type: RoomType = RoomType.NORMAL


@export var room_size_cells := Vector2i(
	41,
	25
)


@export_range(1, 5)
var wall_thickness: int = 2


# ============================================================
# Dungeon coordinate
# ============================================================

var room_coord := Vector2i.ZERO


# ============================================================
# Runtime state
# ============================================================

var encounter_started_once := false

var encounter_active := false

var cleared := false


# 当前存活敌人
var active_enemy_ids: Dictionary = {}


# 当前房间的战斗门
var runtime_doors: Array[RoomDoor] = []


# ============================================================

@onready var ground := (

@onready var grass := (

	_setup_player_detector()

		room_size_cells.x * tile_size.x,
		room_size_cells.y * tile_size.y

# ============================================================

# ============================================================

		return

		player_detector.get_node_or_null(
			"CollisionShape2D"
		)
		as CollisionShape2D
	)


	if collision_shape == null:

		push_error(
			"%s 缺少 PlayerDetector/CollisionShape2D"
			% name
		)

		return


	var tile_size := get_tile_size()


	var rectangle := RectangleShape2D.new()


	rectangle.size = Vector2(

		(
			room_size_cells.x
			- wall_thickness * 2
		)
		* tile_size.x,

		(
			room_size_cells.y
			- wall_thickness * 2
		)
		* tile_size.y
	)


	collision_shape.shape = rectangle


	collision_shape.position = (
		get_room_pixel_size()
		* 0.5
	)


	player_detector.collision_layer = 0

	player_detector.collision_mask = 1


	if not player_detector.body_entered.is_connected(
		_on_body_entered_room
	):

		player_detector.body_entered.connect(
			_on_body_entered_room
		)


# ============================================================
# Enter room
# ============================================================

func _on_body_entered_room(
	body: Node2D
) -> void:

	if not body.is_in_group("player"):
		return


	room_entered.emit(self)


	# 初始房没有战斗
	if room_type == RoomType.START:
		return


	# 已清理
	if cleared:
		return


	# 防止重复触发
	if encounter_started_once:
		return


	encounter_started_once = true


	# 下一帧开始战斗
	call_deferred(
		"_start_encounter"
	)


# ============================================================
# Encounter
# ============================================================

func _start_encounter() -> void:

	if cleared:
		return


	if encounter_active:
		return


	encounter_active = true


	# 先关门
	set_all_doors_locked(
		true
	)


	# 再刷怪
	spawn_room_enemies()


	encounter_started.emit(
		self
	)


	# 防止房间没有可用出生点导致永远锁门
	if active_enemy_ids.is_empty():

		call_deferred(
			"_complete_encounter"
		)


# ============================================================
# 子类负责真正生成敌人
# ============================================================

func spawn_room_enemies() -> void:
	pass


# ============================================================
# Register Enemy
# ============================================================

func register_enemy(
	enemy: Node
) -> void:

	if enemy == null:
		return


	var enemy_id := (
		enemy.get_instance_id()
	)


	active_enemy_ids[
		enemy_id
	] = enemy


	enemy.tree_exited.connect(
		_on_enemy_tree_exited.bind(
			enemy_id
		)
	)


# ============================================================
# Enemy removed
# ============================================================

func _on_enemy_tree_exited(
	enemy_id: int
) -> void:

	active_enemy_ids.erase(
		enemy_id
	)


	if not encounter_active:
		return


	if active_enemy_ids.is_empty():

		call_deferred(
			"_complete_encounter"
		)


# ============================================================
# Clear Room
# ============================================================

func _complete_encounter() -> void:

	if not encounter_active:
		return


	encounter_active = false

	cleared = true


	set_all_doors_locked(
		false
	)


	room_cleared.emit(
		self
	)


# ============================================================
# Doors
# ============================================================

func get_or_create_doors_root() -> Node2D:

	if doors_root:
		return doors_root


	doors_root = Node2D.new()

	doors_root.name = "Doors"

	add_child(
		doors_root
	)


	return doors_root


func register_door(
	door: RoomDoor
) -> void:

	if door == null:
		return


	if runtime_doors.has(
		door
	):
		return


	runtime_doors.append(
		door
	)


	door.set_locked(
		encounter_active
	)


func set_all_doors_locked(
	value: bool
) -> void:

	for door in runtime_doors:

		if is_instance_valid(
			door
		):

			door.set_locked(
				value
			)


# ============================================================
# Debug
# ============================================================

func force_clear_room() -> void:

	for enemy in (
		active_enemy_ids.values()
	):

		if is_instance_valid(enemy):
			enemy.queue_free()


# ============================================================
# Minimap
# ============================================================

func get_open_directions() -> Dictionary:

	return {
		"north": false,
		"south": false,
		"east": false,
		"west": false
	}
```

```text
未进入
    ↓
进入
    ↓

    ↓
关门
    ↓
刷怪
    ↓
encounter_active = true
    ↓
最后一只怪 queue_free()
    ↓

    ↓

    ↓
开门
```

# 4. 给 Player 增加临时攻击

为了让你现在能测试“杀光怪→开门”，先给玩家加一个非常简单的范围攻击。

用下面版替换 `player.gd`：

```gdscript
extends CharacterBody2D

class_name PlayerController


@export_category("Movement")

@export var move_speed: float = 350.0


@export_category("Debug Combat")

@export var debug_attack_enabled := true

@export var debug_attack_range := 130.0

@export var debug_attack_damage := 1

@export var attack_action: StringName = &"ui_accept"


func _ready() -> void:

	add_to_group(
		"player"
	)


	z_index = 10


	# Player
	collision_layer = 1


	# World + Enemy
	collision_mask = 3


	motion_mode = (
		CharacterBody2D.MOTION_MODE_FLOATING
	)


func _physics_process(
	_delta: float
) -> void:

	_handle_movement()

	_handle_debug_attack()


func _handle_movement() -> void:

	var direction := Input.get_vector(
		"ui_left",
		"ui_right",
		"ui_up",
		"ui_down"
	)


	velocity = (
		direction
		* move_speed
	)


	move_and_slide()


# ============================================================
# 临时攻击
#
# 默认 Space / Enter
# ============================================================

func _handle_debug_attack() -> void:

	if not debug_attack_enabled:
		return


	if not Input.is_action_just_pressed(
		attack_action
	):
		return


	var enemies := (
		get_tree()
			.get_nodes_in_group(
				"enemy"
			)
	)


	for enemy in enemies:

		if not enemy is Node2D:
			continue


		var enemy_2d := (
			enemy as Node2D
		)


		var distance := (
			global_position.distance_to(
				enemy_2d.global_position
			)
		)


		if distance > debug_attack_range:
			continue


		if enemy.has_method(
			"take_damage"
		):

			enemy.call(
				"take_damage",
				debug_attack_damage
			)
```

```text
测试敌人 2 HP

```

# 7. 再增加真正刷怪的代码

```gdscript
# ============================================================
# Spawn Enemies
# ============================================================

func spawn_room_enemies() -> void:

	var container := (
		_get_or_create_enemy_container()
	)


	var candidates := (
		_get_enemy_spawn_candidates()
	)


	if candidates.is_empty():

		push_warning(
			"%s 没有有效敌人出生位置"
			% name
		)

		return


	var count := rng.randi_range(

		min_enemies,

		max_enemies
	)


	count = min(
		count,
		candidates.size()
	)


	for i in range(count):

		var random_index := (
			rng.randi_range(
				0,
				candidates.size() - 1
			)
		)


		var cell := (
			candidates[
				random_index
			]
		)


		candidates.remove_at(
			random_index
		)


		var enemy := (
			_create_enemy_instance()
		)


		if enemy == null:
			continue


		container.add_child(
			enemy
		)


		enemy.global_position = (
			cell_to_global(
				cell
			)
		)


		register_enemy(
			enemy
		)


# ============================================================
# Enemy Instance
# ============================================================

func _create_enemy_instance() -> Node2D:

	# 没放正式怪物时
	# 自动用测试敌人

	if enemy_scenes.is_empty():

		return EnemyDummy.new()


	var scene := enemy_scenes[
		rng.randi_range(
			0,
			enemy_scenes.size() - 1
		)
	]


	if scene == null:
		return null


	var instance := (
		scene.instantiate()
	)


	if not instance is Node2D:

		instance.queue_free()

		push_error(
			"Enemy Scene 根节点必须继承 Node2D"
		)

		return null


	return instance as Node2D


# ============================================================
# Spawn Candidates
# ============================================================

func _get_enemy_spawn_candidates() -> Array[Vector2i]:

	var result: Array[Vector2i] = []


	var center := (
		get_center_cell()
	)


	# 额外留两格
	# 避免怪直接生成在墙边

	var padding := (
		wall_thickness + 2
	)


	for x in range(

		padding,

		room_size_cells.x
			- padding

	):

		for y in range(

			padding,

			room_size_cells.y
				- padding

		):

			var cell := (
				Vector2i(x, y)
			)


			# =================================================
			# 有墙 / 柱子的位置不能刷怪
			# =================================================

			if walls.get_cell_source_id(
				cell
			) != -1:

				continue


			# =================================================
			# 玩家出生中心附近不能刷怪
			# =================================================

			var distance := Vector2(
				cell - center
			).length()


			if distance < (
				min_enemy_distance_cells
			):

				continue


			result.append(
				cell
			)


	return result


# ============================================================
# Enemy Container
# ============================================================

func _get_or_create_enemy_container() -> Node2D:

	if enemy_container:
		return enemy_container


	enemy_container = Node2D.new()

	enemy_container.name = (
		"EnemyContainer"
	)


	add_child(
		enemy_container
	)


	return enemy_container
```

这部分有一个很重要的效果：

**不会再直接写：**

```gdscript
Vector2(800, 700)
```

而是：

```gdscript
Vector2i(10, 8)
```

然后通过：

```gdscript
cell_to_global(cell)
```

转换。

所以你的 64×64 Tile 不会再次造成敌人半格错位。

# 8. DungeonGenerator 增加战斗状态同步

在你上一版 `_spawn_rooms()` 中，原本已经有：

```gdscript
room.room_entered.connect(
	_on_room_entered
```

在下面再加：

```gdscript
room.encounter_started.connect(
	_on_encounter_started
)


room.room_cleared.connect(
	_on_room_cleared
)
```

然后在 `DungeonGenerator.gd` 最下面增加：

```gdscript
# ============================================================
# Encounter
# ============================================================

func _on_encounter_started(
	_room: RoomBase
) -> void:

	minimap_updated.emit()


func _on_room_cleared(
	_room: RoomBase
) -> void:

	minimap_updated.emit()
```

这样：

```text
进入战斗
→ 小地图刷新

清理完成
→ 小地图再次刷新
```

# 9. 小地图增加“战斗中”提示

在 `MinimapView.gd` 颜色区增加：

```gdscript
@export var combat_outline_color := Color(
	0.95,
	0.18,
	0.15,
	1.0
)
```

然后找到之前：

```gdscript
draw_rect(
	room_rect,
	outline_color,
	false,
	max(
		1.0,
		2.0 * scale_factor
	)
)
```

替换成：

```gdscript
var room := (
	dungeon.get_room(
		coord
	)
)


var current_outline := (
	outline_color
)


var outline_width := max(
	1.0,
	2.0 * scale_factor
)


# 当前房间处于战斗状态
if (
	room != null
	and room.encounter_active
):

	current_outline = (
		combat_outline_color
	)


	outline_width = max(
		2.0,
		4.0 * scale_factor
	)


draw_rect(

	room_rect,

	current_outline,

	false,

	outline_width
)
```

效果：

```text
黄色房间
= 当前房间

黄色 + 红色粗边
= 当前正在战斗

灰色
= 已探索且已经离开的房间

蓝色
= 当前可去但尚未探索的房间
```
