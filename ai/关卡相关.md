## DungeonGenerator.gd

挂载在 DungeonManager 节点：

```gdscript
extends Node2D
class_name DungeonGenerator


@export var room_scene: PackedScene

@export var dungeon_size := 20
@export var max_rooms := 12

@export var room_distance := 256


var rng := RandomNumberGenerator.new()


var rooms: Dictionary = {}

var directions = [
	Vector2i.UP,
	Vector2i.DOWN,
	Vector2i.LEFT,
	Vector2i.RIGHT
]


enum RoomType
{
	START,
	NORMAL,
	TREASURE,
	BOSS,
	SECRET
}


class RoomData:

	var pos: Vector2i
	var type: RoomType
	var doors = []

	func _init(p):
		pos = p



func generate(seed_value:int=-1):

	if seed_value != -1:
		rng.seed = seed_value
	else:
		rng.randomize()


	rooms.clear()


	create_start_room()

	grow_rooms()

	assign_special_rooms()

	spawn_rooms()



# 创建初始房

func create_start_room():

	var start = RoomData.new(Vector2i.ZERO)

	start.type = RoomType.START

	rooms[start.pos]=start



# 随机扩展房间

func grow_rooms():

	var frontier=[
		Vector2i.ZERO
	]


	while rooms.size()dungeon_size:
			continue

		if abs(next.y)>dungeon_size:
			continue



		var room=RoomData.new(next)

		room.type=RoomType.NORMAL


		rooms[next]=room

		connect_rooms(current,next)


		frontier.append(next)



# 房间连接

func connect_rooms(a:Vector2i,b:Vector2i):

	var room_a=rooms[a]
	var room_b=rooms[b]


	room_a.doors.append(b-a)

	room_b.doors.append(a-b)



# 特殊房间分配

func assign_special_rooms():

	var positions=rooms.keys()


	positions.erase(Vector2i.ZERO)


	if positions.size()<2:
		return



	# 最远房间作为boss

	var boss_pos=find_farthest_room()

	rooms[boss_pos].type=RoomType.BOSS



	# 随机宝箱房

	var treasure_pos=positions.pick_random()

	if treasure_pos!=boss_pos:

		rooms[treasure_pos].type=RoomType.TREASURE





func find_farthest_room():

	var max_dist=-1

	var result=null


	for p in rooms.keys():

		var d=Vector2i.ZERO.distance_to(p)

		if d>max_dist:

			max_dist=d

			result=p


	return result





# 实例化房间

func spawn_rooms():

	for pos in rooms:

		var data=rooms[pos]


		var room=room_scene.instantiate()


		room.position=Vector2(
			pos.x*room_distance,
			pos.y*room_distance
		)


		room.room_type=data.type

		room.doors=data.doors


		add_child(room)
```

继续实现 BSP 地图生成器。

根据《噬铁者》设计文档，地图生成有几个**必须满足的约束**：

1. 使用 **BSP（二叉空间分割）算法**生成房间布局；  
2. 每个房间至少连接一个其他房间；
3. 初始房固定 `(0,0)`，至少连接两个房间；
4. Boss 房固定在可用范围最远端；

# 二、TileMap 自动铺地与墙壁生成

使用 Godot 4 的 `TileMap`（或 `TileMapLayer`）遍历逻辑数据，生成底层的地板和墙壁。

在 `DungeonGenerator.gd` 中添加：

```gdscript
@onready var tilemap: TileMap = $TileMap
@export var theme: ThemeConfig

const TILE_SIZE = 64
const LAYER_FLOOR = 0
const LAYER_WALL = 1

func generate_terrain():
	tilemap.tile_set = theme.tile_set
	tilemap.clear()
	
	for room in room_graph.rooms:
		_carve_room_terrain(room)
		
	for corridor in corridors:
		_carve_corridor_terrain(corridor)

func _carve_room_terrain(room: RoomData):
	# 绘制地板
	for x in range(room.rect.position.x, room.rect.end.x):
		for y in range(room.rect.position.y, room.rect.end.y):
			tilemap.set_cell(LAYER_FLOOR, Vector2i(x, y), 0, Vector2i(0, 0)) # 假设 0,0 是地板 Atlas
			
	# 绘制围墙 (房间边缘向外扩展一格)
	for x in range(room.rect.position.x - 1, room.rect.end.x + 1):
		tilemap.set_cell(LAYER_WALL, Vector2i(x, room.rect.position.y - 1), 0, Vector2i(1, 0)) # 上墙
		tilemap.set_cell(LAYER_WALL, Vector2i(x, room.rect.end.y), 0, Vector2i(1, 0)) # 下墙
		
	for y in range(room.rect.position.y - 1, room.rect.end.y + 1):
		tilemap.set_cell(LAYER_WALL, Vector2i(room.rect.position.x - 1, y), 0, Vector2i(1, 0)) # 左墙
		tilemap.set_cell(LAYER_WALL, Vector2i(room.rect.end.x, y), 0, Vector2i(1, 0)) # 右墙
```

## GPT 5
根据《噬铁者》关卡设计分册中的核心设定，这里有一个极度关键的隐藏细节：**1 个网格 = 1 个屏幕（约 1920×1080）**。

这意味着，房间的尺寸直接决定了相机的行为：
*   **1×1 房间：** 相机边界精确等于 1 个屏幕尺寸，相机**完全锁死**在房间中心（经典《以撒》体验）。
*   **3×3 房间：** 相机边界扩展为 3×3 个屏幕尺寸，相机可以**在房间内部跟随玩家**，但到达房间边缘时会平滑停止。

为了实现这种“动态边界锁定”且兼顾平滑进门切换，我们需要结合 `Area2D` 触发器和 `Camera2D` 的动态 Limit Tween。


  {/* Reason: 此功能需要从底层房间实例到顶层相机的顺序配置，乱序会导致依赖缺失（如找不到引用或信号未连接）。 */}

  
    在每个实例化的房间中，我们需要动态生成一个 `Area2D`。这个碰撞盒的尺寸严格等于该房间的逻辑屏幕尺寸。

    ```gdscript
    extends Node2D
    class_name DungeonRoom

    const SCREEN_SIZE = Vector2(1920, 1080)
    var room_rect: Rect2i

    # 定义全局信号，通知相机切换
    signal player_entered_room(rect: Rect2i)

    func initialize(data: RoomData):
        room_rect = data.rect
        _setup_room_trigger()

    func _setup_room_trigger():
        var area = Area2D.new()
        area.collision_layer = 0
        area.collision_mask = 1 # 假设 Player 在 Layer 1

        var shape_node = CollisionShape2D.new()
        var rect_shape = RectangleShape2D.new()
        
        # 尺寸设为网格数 × 屏幕分辨率
        rect_shape.size = Vector2(room_rect.size.x * SCREEN_SIZE.x, room_rect.size.y * SCREEN_SIZE.y)
        shape_node.shape = rect_shape
        
        # 调整形状到房间本地中心
        shape_node.position = rect_shape.size / 2.0
        
        area.add_child(shape_node)
        add_child(area)
        
        area.body_entered.connect(_on_body_entered)

    func _on_body_entered(body: Node2D):
        if body.is_in_group("player"):
            player_entered_room.emit(room_rect)
    ```
  

  
    将 `Camera2D` 作为 `Player` 的子节点，并在编辑器中确保开启 `Position Smoothing`，让相机跟随玩家时具备基础的平滑感。

    新建一个脚本 `RoomCamera.gd` 挂载在玩家的 `Camera2D` 上：

    ```gdscript
    extends Camera2D
    class_name RoomCamera

    const SCREEN_SIZE = Vector2(1920, 1080)
    var current_tween: Tween

    func _ready():
        # 确保平滑开启
        position_smoothing_enabled = true
        position_smoothing_speed = 5.0

    func switch_to_room(room_rect: Rect2i):
        # 计算新房间的物理边界（世界坐标系）
        var target_left = room_rect.position.x * SCREEN_SIZE.x
        var target_top = room_rect.position.y * SCREEN_SIZE.y
        var target_right = room_rect.end.x * SCREEN_SIZE.x
        var target_bottom = room_rect.end.y * SCREEN_SIZE.y

        if current_tween and current_tween.is_valid():
            current_tween.kill()

        # 使用 Tween 平滑过渡相机的 Limit，营造“推镜头”感
        current_tween = create_tween().set_parallel(true).set_trans(Tween.TRANS_SINE).set_ease(Tween.EASE_OUT)
        
        var duration = 0.6 # 过渡时间，可调整
        
        current_tween.tween_property(self, "limit_left", target_left, duration)
        current_tween.tween_property(self, "limit_top", target_top, duration)
        current_tween.tween_property(self, "limit_right", target_right, duration)
        current_tween.tween_property(self, "limit_bottom", target_bottom, duration)
    ```
  

  
    最后，在你的 `DungeonManager` 或生成器的 `spawn_room_scenes` 环节，捕获房间发出的信号，并将其转发给玩家相机。

    func spawn_room_scenes():

在 Godot 4 中，动态地图的寻路烘焙比以往版本要简单且强大得多。因为我们使用了 `TileMap` 铺设墙壁，并通过 `PackedScene` 生成带有内部障碍物（如岩石、深渊）的房间，最佳方案是**使用 `NavigationRegion2D` 的运行时几何解析（Runtime Geometry Parsing）进行自动烘焙**。

以下是完整的实现流程：



{/* Reason: 烘焙寻路需要从节点层级配置到代码触发，再到敌人使用的完整流程，顺序至关重要。 */}

  
    在你的地牢生成器节点下，添加一个 `NavigationRegion2D` 节点，并将需要参与寻路烘焙的物理节点（TileMap、房间、门）作为它的子节点。

    这样做的目的是让它默认解析子节点的几何形状（静态碰撞体）。

    ```text
    DungeonGenerator (Node2D)
    └── NavigationRegion2D (NavigationRegion2D)
        ├── TileMap (TileMap)   # 包含地板与墙壁碰撞
        ├── Rooms (Node2D)      # 实例化的各种房间
        └── Doors (Node2D)      # 实例化的门
    ```
  

  
    选中 `NavigationRegion2D` 节点，在属性面板中：
    1. 找到 **NavigationPolygon** 属性，新建一个 `NavigationPolygon` 资源。
    2. 展开它，找到 **Parsed Geometry Type**，将其设置为 **Static Colliders**（静态碰撞体）。这会让它读取所有带有碰撞层的节点。
    3. 找到 **Agent Radius**，设置一个合理的值（比如 `32.0`），这是敌人的碰撞半径。留出这个边距可以防止敌人贴墙走时卡住。
  

  
    回到你的 `DungeonGenerator.gd`，在物理场景生成完毕后，调用烘焙函数。由于烘焙是异步操作，我们需要等待烘焙完成的信号，然后再通知游戏正式开始（或开始刷怪）。

    ```gdscript
    @onready var nav_region: NavigationRegion2D = $NavigationRegion2D

    func generate_layer(layer_config):
        # 1. 逻辑生成
        generate_bsp(layer_config.map_size)
        
        # 2. 物理生成
        generate_terrain()
        spawn_room_scenes()
        
        # 3. 烘焙寻路
        bake_navmesh()

    func bake_navmesh():
        # 触发烘焙
        nav_region.bake_navigation_polygon()
        
        # 等待烘焙完成的信号
        await nav_region.bake_finished
        print("寻路网格烘焙完成！")
        
        # 通知系统可以开始刷怪或解锁玩家控制了
        start_gameplay()
    ```
  

  
    在敌人的 `PackedScene` 中，添加一个 `NavigationAgent2D` 节点。配置其 `Target Desired Distance`（到达目标的容差距离，如 20px）。

    在敌人的脚本中，编写追踪逻辑：

    ```gdscript
    extends CharacterBody2D
    class_name Enemy

    @onready var nav_agent: NavigationAgent2D = $NavigationAgent2D
    @export var speed: float = 150.0

    var player: Node2D

    func _ready():
        # 获取玩家引用
        player = get_tree().get_first_node_in_group("player")
        # 设定寻路更新频率避免性能问题（可选，例如用 Timer 触发）

    func _physics_process(delta):
        if not player: return
        
        # 1. 设定目标位置为玩家当前位置
        nav_agent.target_position = player.global_position
        
        # 2. 如果已经到达目标范围，则停止
        if nav_agent.is_navigation_finished():
            velocity = Vector2.ZERO
            move_and_slide()
            return
            
        # 3. 获取寻路路径上的下一个目标点
        var next_path_position: Vector2 = nav_agent.get_next_path_position()
        
        # 4. 计算移动方向并应用速度
        var direction = global_position.direction_to(next_path_position)
        velocity = direction * speed
        
        move_and_slide()
    ```
  



### 避坑指南：
1. **TileMap 物理层检查：** 确保你的墙壁 Tile 配置了 Physics Layer 并在对应的图块上画了碰撞多边形，否则 `NavigationRegion2D` 无法识别墙壁，敌人会穿墙。
2. **只解析静态碰撞体：** 玩家和敌人通常使用 `CharacterBody2D`（动态物体），而 `Parsed Geometry Type` 设为 `Static Colliders` 可以自动忽略这些动态物体，避免烘焙时把玩家自己也当作障碍物抠掉。
3. **门的状态更新：** 如果你的门有开/关状态，并在关闭时具有 `StaticBody2D` 碰撞。当地图清空怪物门打开时，你可以禁用门的碰撞体，并**再次调用** `nav_region.bake_navigation_polygon()` 更新寻路网格，这样敌人就能跨越房间追击了。
