### 2. 初始房代码

把下面脚本挂在 `StartRoom_01` 根节点：

```gdscript
extends Node2D

class_name StartRoom01


# =========================================================
# 《噬铁者》第一层初始房
# StartRoom_01
#
# 房间规格：
# 30 x 18 Tile
# Tile：64 x 64 px
# 墙体厚度：2 Tile
#
# 可活动区：
# 26 x 14 Tile
#
# 功能：
# - 初始安全区
# - 中央出生
# - 中央暖色光照
# - 四周草地
# - 南侧入口
# - 北侧出口
# =========================================================


@export_category("Room Size")

@export var room_width: int = 30
@export var room_height: int = 18
@export var wall_thickness: int = 2


@export_category("TileSet")

# TileSet 中的 Source ID
@export var tile_source_id: int = 0


# ---------- 地板 ----------

@export var floor_tile: Vector2i = Vector2i(0, 0)


# ---------- 墙 ----------

@export var wall_tile: Vector2i = Vector2i(1, 0)


# ---------- 草 ----------

@export var grass_tiles: Array[Vector2i] = [
	Vector2i(0, 1),
	Vector2i(1, 1),
	Vector2i(2, 1),
	Vector2i(3, 1)
]


@export_category("Grass")

# 草地向房间内部延伸多少格
@export_range(1, 6)
var grass_depth: int = 4


# 草地密度
@export_range(0.0, 1.0)
var grass_density: float = 0.65


# 中心区域保持无草
@export var center_clear_radius: float = 5.0


@export_category("Lighting")

@export var center_light_energy: float = 1.4

@export var center_light_scale: float = 5.0


# =========================================================
# Nodes
# =========================================================

@onready var ground: TileMapLayer = $Ground
@onready var grass: TileMapLayer = $Grass
@onready var walls: TileMapLayer = $Walls

@onready var center_light: PointLight2D = $CenterLight

@onready var player_spawn: Marker2D = $PlayerSpawn
@onready var entrance_south: Marker2D = $EntranceSouth
@onready var exit_north: Marker2D = $ExitNorth


var rng := RandomNumberGenerator.new()


# =========================================================
# Ready
# =========================================================

func _ready() -> void:

	rng.randomize()

	generate_room()

	setup_spawn_points()

	setup_light()


# =========================================================
# Generate Room
# =========================================================

func generate_room() -> void:

	ground.clear()
	grass.clear()
	walls.clear()

	generate_floor()
	generate_walls()
	generate_grass()


# =========================================================
# Floor
# =========================================================

func generate_floor() -> void:

	for x in range(room_width):

		for y in range(room_height):

			var cell := Vector2i(x, y)

			ground.set_cell(
				cell,
				tile_source_id,
				floor_tile
			)


# =========================================================
# Walls
# =========================================================

func generate_walls() -> void:

	for x in range(room_width):

		for y in range(room_height):

			var is_left_wall: bool = x < wall_thickness
			var is_right_wall: bool = x >= room_width - wall_thickness

			var is_top_wall: bool = y < wall_thickness
			var is_bottom_wall: bool = y >= room_height - wall_thickness


			if (
				is_left_wall
				or is_right_wall
				or is_top_wall
				or is_bottom_wall
			):

				var cell := Vector2i(x, y)

				walls.set_cell(
					cell,
					tile_source_id,
					wall_tile
				)


	# -----------------------------------------------------
	# 南侧入口
	# 开 3 Tile 宽
	# -----------------------------------------------------

	var center_x := room_width / 2

	for x in range(center_x - 1, center_x + 2):

		for y in range(
			room_height - wall_thickness,
			room_height
		):

			walls.erase_cell(
				Vector2i(x, y)
			)


	# -----------------------------------------------------
	# 北侧出口
	# 开 3 Tile 宽
	# -----------------------------------------------------

	for x in range(center_x - 1, center_x + 2):

		for y in range(0, wall_thickness):

			walls.erase_cell(
				Vector2i(x, y)
			)


# =========================================================
# Grass
# =========================================================

func generate_grass() -> void:

	if grass_tiles.is_empty():
		return


	var center := Vector2(
		room_width / 2.0,
		room_height / 2.0
	)


	for x in range(
		wall_thickness,
		room_width - wall_thickness
	):

		for y in range(
			wall_thickness,
			room_height - wall_thickness
		):

			var cell := Vector2i(x, y)


			# ---------------------------------------------
			# 与墙的距离
			# ---------------------------------------------

			var distance_left := x - wall_thickness

			var distance_right := (
				room_width
				- wall_thickness
				- 1
				- x
			)

			var distance_top := y - wall_thickness

			var distance_bottom := (
				room_height
				- wall_thickness
				- 1
				- y
			)


			var edge_distance := min(
				distance_left,
				distance_right,
				distance_top,
				distance_bottom
			)


			# 只让草长在外围
			if edge_distance >= grass_depth:
				continue


			# ---------------------------------------------
			# 中央区域不长草
			# ---------------------------------------------

			var distance_to_center := Vector2(
				x,
				y
			).distance_to(center)


			if distance_to_center < center_clear_radius:
				continue


			# ---------------------------------------------
			# 越靠近墙草越密
			# ---------------------------------------------

			var edge_factor := 1.0 - (
				float(edge_distance)
				/ float(grass_depth)
			)


			var probability := (
				grass_density
				* edge_factor
			)


			if rng.randf() > probability:
				continue


			var tile := grass_tiles[
				rng.randi_range(
					0,
					grass_tiles.size() - 1
				)
			]


			grass.set_cell(
				cell,
				tile_source_id,
				tile
			)


# =========================================================
# Spawn / Door Positions
# =========================================================

func setup_spawn_points() -> void:

	var tile_size := 64.0


	var room_center := Vector2(
		room_width * tile_size * 0.5,
		room_height * tile_size * 0.5
	)


	# 玩家出生在中心稍微偏下
	player_spawn.position = (
		room_center
		+ Vector2(0, 96)
	)


	# 南入口
	entrance_south.position = Vector2(
		room_center.x,
		(room_height - wall_thickness) * tile_size
	)


	# 北出口
	exit_north.position = Vector2(
		room_center.x,
		wall_thickness * tile_size
	)


	# 中心灯
	center_light.position = room_center


# =========================================================
# Lighting
# =========================================================

func setup_light() -> void:

	center_light.energy = center_light_energy

	center_light.texture_scale = center_light_scale
```

### 3. 中央光照

```text
中心：
████████
████████
████████

中间：
▓▓▓▓▓▓▓▓

外围：
▒▒▒▒▒▒▒▒

边缘：
透明
```

也就是：

```text
白色

浅灰
 ↓
深灰

黑色/透明
```

```text
Energy = 1.4

Texture Scale = 5.0

Blend Mode = Add

Shadow Enabled = true
```

# 三、核心脚本 room_base.gd

保存为：

```text
res://rooms/base/room_base.gd
```

代码：

```gdscript
extends Node2D

class_name RoomBase


# ============================================================
# 《噬铁者》统一房间基类
#
# 负责：
# 1. 玩家进入检测
# 2. 战斗开始
# 3. 门锁定
# 4. 敌人数量管理
# 5. 清怪判定
# 6. 房间完成
# 7. 出口请求
# ============================================================


signal room_entered(room: RoomBase)

signal encounter_started(room: RoomBase)

signal room_cleared(room: RoomBase)

signal exit_requested(
	room: RoomBase,
	door: RoomDoor
)


# ============================================================
# Room Type
# ============================================================

enum RoomType {

	START,

	NORMAL,

	ELITE,

	SPECIAL,

	BOSS
}


@export_category("Room")

@export var room_id: StringName = &"ROOM_000"

@export var room_type: RoomType = RoomType.NORMAL


# 初始房等安全房直接视为完成
@export var starts_cleared: bool = false


# 战斗开始后是否锁门
@export var lock_doors_during_combat: bool = true


# ============================================================
# Node References
# ============================================================

@onready var ground: TileMapLayer = get_node_or_null("Ground")

@onready var grass: TileMapLayer = get_node_or_null("Grass")

@onready var walls: TileMapLayer = get_node_or_null("Walls")


@onready var enemy_container: Node2D = (
	get_node_or_null("EnemyContainer")
)


@onready var spawn_points: Node2D = (
	get_node_or_null("SpawnPoints")
)


@onready var doors_root: Node2D = (
	get_node_or_null("Doors")
)


@onready var player_detector: Area2D = (
	get_node_or_null("PlayerDetector")
)


@onready var player_spawn: Marker2D = (
	get_node_or_null("PlayerSpawn")
)


# ============================================================
# Runtime State
# ============================================================

var has_player_entered: bool = false

var encounter_active: bool = false

var cleared: bool = false


# 使用 Instance ID 管理敌人
var active_enemy_ids: Dictionary = {}


# ============================================================
# Ready
# ============================================================

func _ready() -> void:

	_connect_player_detector()

	_connect_doors()


	if starts_cleared or room_type == RoomType.START:

		cleared = true

		_set_all_doors_locked(false)

	else:

		cleared = false

		# 玩家没进入以前允许进入
		_set_all_doors_locked(false)


# ============================================================
# Player Detection
# ============================================================

func _connect_player_detector() -> void:

	if player_detector == null:
		return


	if not player_detector.body_entered.is_connected(
		_on_player_detector_body_entered
	):

		player_detector.body_entered.connect(
			_on_player_detector_body_entered
		)


func _on_player_detector_body_entered(
	body: Node2D
) -> void:

	if not body.is_in_group("player"):
		return


	if has_player_entered:
		return


	has_player_entered = true


	room_entered.emit(self)


	# 初始房 / 已清理房不进入战斗
	if cleared:
		return


	start_encounter()


# ============================================================
# Encounter
# ============================================================

func start_encounter() -> void:

	if encounter_active:
		return


	if cleared:
		return


	encounter_active = true


	if lock_doors_during_combat:

		_set_all_doors_locked(true)


	encounter_started.emit(self)


	# 子房间实现具体刷怪
	_spawn_room_enemies()


	# 注册已经生成的敌人
	_register_existing_enemies()


	# 如果没有怪，则直接完成
	if active_enemy_ids.is_empty():

		call_deferred("finish_room")


# ============================================================
# 子类覆盖
# ============================================================

func _spawn_room_enemies() -> void:

	pass


# ============================================================
# Enemy Register
# ============================================================

func register_enemy(enemy: Node) -> void:

	if enemy == null:
		return


	var enemy_id := enemy.get_instance_id()


	if active_enemy_ids.has(enemy_id):
		return


	active_enemy_ids[enemy_id] = true


	enemy.tree_exited.connect(
		_on_enemy_tree_exited.bind(enemy_id),
		CONNECT_ONE_SHOT
	)


# ============================================================

func _register_existing_enemies() -> void:

	if enemy_container == null:
		return


	for enemy in enemy_container.get_children():

		register_enemy(enemy)


# ============================================================

func _on_enemy_tree_exited(
	enemy_id: int
) -> void:

	active_enemy_ids.erase(enemy_id)


	if (
		encounter_active
		and active_enemy_ids.is_empty()
	):

		call_deferred("finish_room")


# ============================================================
# Room Complete
# ============================================================

func finish_room() -> void:

	if cleared:
		return


	cleared = true

	encounter_active = false


	_set_all_doors_locked(false)


	room_cleared.emit(self)


	_on_room_cleared()


# ============================================================
# 子类可覆盖
# ============================================================

func _on_room_cleared() -> void:

	pass


# ============================================================
# Doors
# ============================================================

func _connect_doors() -> void:

	if doors_root == null:
		return


	for node in doors_root.get_children():

		if node is RoomDoor:

			var door := node as RoomDoor


			if not door.used.is_connected(
				_on_door_used
			):

				door.used.connect(
					_on_door_used
				)


# ============================================================

func _set_all_doors_locked(
	value: bool
) -> void:

	if doors_root == null:
		return


	for node in doors_root.get_children():

		if node is RoomDoor:

			node.set_locked(value)


# ============================================================

func _on_door_used(
	door: RoomDoor
) -> void:

	if not cleared:
		return


	exit_requested.emit(
		self,
		door
	)


# ============================================================
# Spawn
# ============================================================

func get_player_spawn_position() -> Vector2:

	if player_spawn:

		return player_spawn.global_position


	return global_position
```

```gdscript
# ============================================================
# Doors
# ============================================================

func carve_doors() -> void:

	var center_x: int = room_width / 2

	var center_y: int = room_height / 2

	var half_door: int = door_width / 2


	# NORTH

	if north_door:

		for x in range(
			center_x - half_door,
			center_x + half_door + 1
		):

			for y in range(wall_thickness):

				walls.erase_cell(
					Vector2i(x, y)
				)


	# SOUTH

	if south_door:

		for x in range(
			center_x - half_door,
			center_x + half_door + 1
		):

			for y in range(
				room_height - wall_thickness,
				room_height
			):

				walls.erase_cell(
					Vector2i(x, y)
				)


	# WEST

	if west_door:

		for y in range(
			center_y - half_door,
			center_y + half_door + 1
		):

			for x in range(wall_thickness):

				walls.erase_cell(
					Vector2i(x, y)
				)


	# EAST

	if east_door:

		for y in range(
			center_y - half_door,
			center_y + half_door + 1
		):

			for x in range(
				room_width - wall_thickness,
				room_width
			):

				walls.erase_cell(
					Vector2i(x, y)
				)


# ============================================================
# Grass / Moss
# ============================================================

func generate_grass() -> void:

	if grass_tiles.is_empty():

		return


	for x in range(
		wall_thickness,
		room_width - wall_thickness
	):

		for y in range(
			wall_thickness,
			room_height - wall_thickness
		):

			var cell := Vector2i(x, y)


			# 墙体上不生成

			if walls.get_cell_source_id(cell) != -1:

				continue


			var left_distance: int = (
				x - wall_thickness
			)

			var right_distance: int = (
				room_width
				- wall_thickness
				- 1
				- x
			)

			var top_distance: int = (
				y - wall_thickness
			)

			var bottom_distance: int = (
				room_height
				- wall_thickness
				- 1
				- y
			)


			var edge_distance: int = min(
				left_distance,
				right_distance,
				top_distance,
				bottom_distance
			)


			if edge_distance >= grass_edge_depth:

				continue


			var edge_factor: float = (

				1.0
				- float(edge_distance)
				/ float(grass_edge_depth)
			)


			var probability: float = (

				grass_density
				* edge_factor
			)


			if rng.randf() > probability:

				continue


			var tile_index := rng.randi_range(
				0,
				grass_tiles.size() - 1
			)


			grass.set_cell(

				cell,

				tile_source_id,

				grass_tiles[tile_index]
			)


# ============================================================
# Enemy Spawn
# ============================================================

func generate_spawn_points() -> void:

	if spawn_points == null:

		return


	# 删除之前自动创建的点

	for child in spawn_points.get_children():

		if child.name.begins_with("AutoSpawn"):

			child.queue_free()


	var tile_size := Vector2(64, 64)


	for i in range(
		enemy_spawn_cells.size()
	):

		var marker := Marker2D.new()


		marker.name = (
			"AutoSpawn%02d"
			% (i + 1)
		)


		spawn_points.add_child(marker)


		var cell := enemy_spawn_cells[i]


		marker.position = Vector2(

			(cell.x + 0.5)
				* tile_size.x,

			(cell.y + 0.5)
				* tile_size.y
		)


# ============================================================
# RoomBase Override
# ============================================================

func _spawn_room_enemies() -> void:

	if enemy_pool.is_empty():

		return


	if spawn_points == null:

		return


	var points: Array[Marker2D] = []


	for child in spawn_points.get_children():

		if child is Marker2D:

			if child.name.begins_with(
				"AutoSpawn"
			):

				points.append(
					child
				)


	if points.is_empty():

		return


	var target_count := rng.randi_range(

		min_enemies,

		max_enemies
	)


	target_count = min(

		target_count,

		points.size()
	)


	# 随机选择出生点

	for i in range(target_count):

		var random_index := rng.randi_range(
			0,
			points.size() - 1
		)


		var spawn_point := points[
			random_index
		]


		points.remove_at(
			random_index
		)


		var enemy_scene := enemy_pool[
			rng.randi_range(
				0,
				enemy_pool.size() - 1
			)
		]


		spawn_enemy(
			enemy_scene,
			spawn_point
		)


# ============================================================

# ============================================================

func spawn_enemy(
	scene: PackedScene,
	spawn_point: Marker2D
) -> void:

	if scene == null:

		return


	if enemy_container == null:

		return


	var enemy := scene.instantiate()


	enemy_container.add_child(
		enemy
	)


	if enemy is Node2D:

		enemy.global_position = (
			spawn_point.global_position
		)


	register_enemy(
		enemy
	)
```
